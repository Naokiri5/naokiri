#!/usr/bin/env bash
set -euo pipefail

SKIP_TESTS=false
for arg in "$@"; do
  case "$arg" in
    --skip-tests)
      SKIP_TESTS=true
      ;;
    *)
      echo "Unknown argument: $arg" >&2
      echo "Usage: $0 [--skip-tests]" >&2
      exit 1
      ;;
  esac
done

# 1. Verify repository root structure
if [[ ! -d "server" || ! -d "web" ]]; then
  echo "ERROR: Must be executed from the root of Naokiri5/naokiri (expected ./server and ./web directories)." >&2
  exit 1
fi

TIMESTAMP="$(date -u +%Y%m%dT%H%M%SZ)"
TMP_STAGE="$(mktemp)"
trap 'rm -f "$TMP_STAGE"' EXIT

CREATED_COUNT=0
UPDATED_COUNT=0
UNCHANGED_COUNT=0

compute_sha256() {
  local file="$1"
  if command -v sha256sum >/dev/null 2>&1; then
    sha256sum "$file" | awk '{print $1}'
  else
    shasum -a 256 "$file" | awk '{print $1}'
  fi
}

# Idempotent atomic writer: compares SHA-256 checksums and creates timestamped backups only on diff
sync_file() {
  local id="$1"
  local target_path="$2"

  cat > "$TMP_STAGE"
  mkdir -p "$(dirname "$target_path")"

  if [[ -f "$target_path" ]]; then
    local existing_hash new_hash
    existing_hash="$(compute_sha256 "$target_path")"
    new_hash="$(compute_sha256 "$TMP_STAGE")"

    if [[ "$existing_hash" == "$new_hash" ]]; then
      printf "  [%-10s] UNCHANGED  -> %s\n" "$id" "$target_path"
      UNCHANGED_COUNT=$((UNCHANGED_COUNT + 1))
      return 0
    fi

    cp -p "$target_path" "${target_path}.bak.${TIMESTAMP}"
    cp "$TMP_STAGE" "$target_path"
    printf "  [%-10s] UPDATED    -> %s (backup: %s.bak.%s)\n" "$id" "$target_path" "$target_path" "$TIMESTAMP"
    UPDATED_COUNT=$((UPDATED_COUNT + 1))
  else
    cp "$TMP_STAGE" "$target_path"
    printf "  [%-10s] CREATED    -> %s\n" "$id" "$target_path"
    CREATED_COUNT=$((CREATED_COUNT + 1))
  fi
}

echo "============================================================================"
echo " MIA 5.0 Dual-Engine Telemetry, OOM Supervisor & CI Bootstrap (Naokiri5/naokiri)"
echo "============================================================================"

export const REFLEX_BUDGET_NS = 10_000_000n;       // 10 ms hard execution ceiling
export const REFLEX_STALL_TIMEOUT_NS = 50_000_000n; // 50 ms dead-worker tripwire
export const REFLEX_WARMUP_TICKS = 5_000n;          // Reset JIT watermarks after warmup

export const enum ReflexWorkerState {
  INIT = 0n,
  WARMING_UP = 1n,
  HEALTHY = 2n,
  DEGRADED_LATENCY = 3n,
  HALTED = 4n,
}

export const enum ReflexSlot {
  // Cache Line 0 (bytes 0..63): Liveness, Heartbeats & Breach Counters
  HEARTBEAT_SEQ = 0,            // Monotonic counter incremented via Atomics.add
  LAST_HEARTBEAT_NS = 1,        // process.hrtime.bigint() timestamp
  WORKER_STATE = 2,             // ReflexWorkerState enum
  TICK_COUNT = 3,               // Total frames evaluated
  BREACH_COUNT = 4,             // Post-warmup frames exceeding 10 ms
  RESERVED_0 = 5,
  RESERVED_1 = 6,
  RESERVED_2 = 7,

  // Cache Line 1 (bytes 64..127): Nanosecond Latency Breakdown
  WIRE_TO_DECISION_LAST_NS = 8, // ingressTs -> decision exit
  WIRE_TO_DECISION_MAX_NS = 9,
  WAKEUP_JITTER_LAST_NS = 10,   // ingressTs -> worker wakeup (SAB scheduling lag)
  COMPUTE_ONLY_LAST_NS = 11,    // worker wakeup -> decision exit
  COMPUTE_ONLY_MAX_NS = 12,
  ELD_P99_NS = 13,              // Event-loop delay p99 (1 ms resolution)
  ELD_MAX_NS = 14,              // Event-loop delay max
  RESERVED_3 = 15,
}

export const TELEMETRY_SAB_BYTES = 16 * BigInt64Array.BYTES_PER_ELEMENT; // 128 bytes

export function createReflexTelemetryBuffer(): SharedArrayBuffer {
  return new SharedArrayBuffer(TELEMETRY_SAB_BYTES);

import { monitorEventLoopDelay, type IntervalHistogram } from 'node:perf_hooks';
import {
  REFLEX_BUDGET_NS,
  REFLEX_WARMUP_TICKS,
  ReflexSlot,
  ReflexWorkerState,
} from '../telemetry/FastReflexTelemetryLayout.js';

export class FastReflexProfiler {
  private readonly view: BigInt64Array;
  private readonly eld: IntervalHistogram;
  private flushTimer: NodeJS.Timeout | null = null;
  private warmupCleared = false;

  constructor(telemetrySab: SharedArrayBuffer) {
    this.view = new BigInt64Array(telemetrySab);
    this.eld = monitorEventLoopDelay({ resolution: 1 }); // 1 ms resolution in ns
  }

  public start(): void {
    this.eld.enable();
    Atomics.store(this.view, ReflexSlot.WORKER_STATE, ReflexWorkerState.WARMING_UP);
    this.pulseHeartbeat();

    // Keep timer referenced when worker awaits Atomics.waitAsync as sole async handle
    this.flushTimer = setInterval(() => {
      this.pulseHeartbeat();

      const rawP99 = this.eld.percentile(99);
      const rawMax = this.eld.max;
      const p99 = Number.isFinite(rawP99) ? BigInt(Math.round(rawP99)) : 0n;
      const max = Number.isFinite(rawMax) ? BigInt(Math.round(rawMax)) : 0n;

      Atomics.store(this.view, ReflexSlot.ELD_P99_NS, p99);
      this.updateMaxWatermark(ReflexSlot.ELD_MAX_NS, max);

      if (this.warmupCleared && p99 > REFLEX_BUDGET_NS) {
        Atomics.store(
          this.view,
          ReflexSlot.WORKER_STATE,
          ReflexWorkerState.DEGRADED_LATENCY,
        );
      }

      this.eld.reset();
    }, 250);
  }

  public stop(): void {
    if (this.flushTimer) {
      clearInterval(this.flushTimer);
      this.flushTimer = null;
    }
    this.eld.disable();
    Atomics.store(this.view, ReflexSlot.WORKER_STATE, ReflexWorkerState.HALTED);
  }

  /**
   * Call immediately after System 1 finishes evaluating a ring-buffer frame
   * (Base DEX liquidity-pull check, mempool snipe filter, or CEX L2 trigger).
   */
  public recordDecision(ingressTsNs: bigint, computeStartNs: bigint): void {
    const nowNs = process.hrtime.bigint();
    const wireToDecisionNs = nowNs - ingressTsNs;
    const wakeupJitterNs = computeStartNs - ingressTsNs;
    const computeOnlyNs = nowNs - computeStartNs;

    Atomics.store(this.view, ReflexSlot.LAST_HEARTBEAT_NS, nowNs);
    Atomics.add(this.view, ReflexSlot.HEARTBEAT_SEQ, 1n);
    const prevTicks = Atomics.add(this.view, ReflexSlot.TICK_COUNT, 1n);

    // Clear JIT warmup spikes once steady-state is reached
    if (!this.warmupCleared && prevTicks + 1n >= REFLEX_WARMUP_TICKS) {
      this.warmupCleared = true;
      Atomics.store(this.view, ReflexSlot.WIRE_TO_DECISION_MAX_NS, 0n);
      Atomics.store(this.view, ReflexSlot.COMPUTE_ONLY_MAX_NS, 0n);
      Atomics.store(this.view, ReflexSlot.ELD_MAX_NS, 0n);
      Atomics.store(this.view, ReflexSlot.WORKER_STATE, ReflexWorkerState.HEALTHY);
    }

    Atomics.store(this.view, ReflexSlot.WIRE_TO_DECISION_LAST_NS, wireToDecisionNs);
    Atomics.store(this.view, ReflexSlot.WAKEUP_JITTER_LAST_NS, wakeupJitterNs);
    Atomics.store(this.view, ReflexSlot.COMPUTE_ONLY_LAST_NS, computeOnlyNs);

    this.updateMaxWatermark(ReflexSlot.WIRE_TO_DECISION_MAX_NS, wireToDecisionNs);
    this.updateMaxWatermark(ReflexSlot.COMPUTE_ONLY_MAX_NS, computeOnlyNs);

    if (this.warmupCleared && wireToDecisionNs > REFLEX_BUDGET_NS) {
      Atomics.add(this.view, ReflexSlot.BREACH_COUNT, 1n);
      Atomics.store(
        this.view,
        ReflexSlot.WORKER_STATE,
        ReflexWorkerState.DEGRADED_LATENCY,
      );
    }
  }

  public pulseHeartbeat(): void {
    Atomics.store(this.view, ReflexSlot.LAST_HEARTBEAT_NS, process.hrtime.bigint());
    Atomics.add(this.view, ReflexSlot.HEARTBEAT_SEQ, 1n);
  }

  public resetWatermarks(): void {
    Atomics.store(this.view, ReflexSlot.WIRE_TO_DECISION_MAX_NS, 0n);
    Atomics.store(this.view, ReflexSlot.COMPUTE_ONLY_MAX_NS, 0n);
    Atomics.store(this.view, ReflexSlot.ELD_MAX_NS, 0n);
    Atomics.store(this.view, ReflexSlot.BREACH_COUNT, 0n);
    if (this.warmupCleared) {
      Atomics.store(this.view, ReflexSlot.WORKER_STATE, ReflexWorkerState.HEALTHY);
    }
  }

  /**
   * Lock-free compare-and-swap (CAS) max watermark update.
   */
  private updateMaxWatermark(slot: ReflexSlot, candidateNs: bigint): void {
    let current = Atomics.load(this.view, slot);
    while (candidateNs > current) {
      const prev = Atomics.compareExchange(this.view, slot, current, candidateNs);
      if (prev === current) break;
      current = prev;
    }
  }
}

  stalled: boolean;
  warmingUp: boolean;
  state: bigint;

  heartbeatAgeMs: number;
  tickCount: bigint;

export function readReflexTelemetry(telemetrySab: SharedArrayBuffer): ReflexHealthSnapshot {

  const nowNs = process.hrtime.bigint();

  const lastHeartbeatNs = Atomics.load(view, ReflexSlot.LAST_HEARTBEAT_NS);
  const state = Atomics.load(view, ReflexSlot.WORKER_STATE);
  const heartbeatSeq = Atomics.load(view, ReflexSlot.HEARTBEAT_SEQ);
  const tickCount = Atomics.load(view, ReflexSlot.TICK_COUNT);
  const breachCount = Atomics.load(view, ReflexSlot.BREACH_COUNT);

  const wireMaxNs = Atomics.load(view, ReflexSlot.WIRE_TO_DECISION_MAX_NS);
  const jitterLastNs = Atomics.load(view, ReflexSlot.WAKEUP_JITTER_LAST_NS);
  const computeLastNs = Atomics.load(view, ReflexSlot.COMPUTE_ONLY_LAST_NS);
  const computeMaxNs = Atomics.load(view, ReflexSlot.COMPUTE_ONLY_MAX_NS);
  const eldP99Ns = Atomics.load(view, ReflexSlot.ELD_P99_NS);

    (state === ReflexWorkerState.HEALTHY || warmingUp) &&

    healthy,
    stalled,
    warmingUp,
    state,
    heartbeatSeq,
    heartbeatAgeMs: Number(ageNs) / NS_PER_MS,

    wireToDecisionLastMs: Number(wireLastNs) / NS_PER_MS,

import { EventEmitter } from 'node:events';
import type { OpsWebhookNotifier } from '../alerts/OpsWebhookNotifier.js';

export type System1TripReason =
  | 'SYSTEM1_HEARTBEAT_LOSS'
  | 'SYSTEM1_LATENCY_DEGRADED'
  | 'SYSTEM1_WORKER_EXIT';

export interface System1BreakerState {
  tripped: boolean;
  reason: System1TripReason | null;
  trippedAt: number | null;
  details: Record<string, unknown>;
}

export type System2TripReason =
  | 'SYSTEM2_OOM_CRASH'
  | 'SYSTEM2_WORKER_EXIT'
  | 'SYSTEM2_MAX_RESPAWNS_EXCEEDED';

export interface System2BreakerState {
  tripped: boolean;
  reason: System2TripReason | null;
  trippedAt: number | null;
  details: Record<string, unknown>;
}

export class UnifiedRiskGovernor extends EventEmitter {
  private dailyDrawdownHalted = false;

  private system1Breaker: System1BreakerState = {
    tripped: false,
    reason: null,
    trippedAt: null,
    details: {},
  };

  private system2Breaker: System2BreakerState = {
    tripped: false,
    reason: null,
    trippedAt: null,
    details: {},
  };

  public isDailyDrawdownHalted(): boolean {
    return this.dailyDrawdownHalted;
  }

  public setDailyDrawdownHalted(halted: boolean): void {
    this.dailyDrawdownHalted = halted;
  }

  // --- System 1 Circuit Breaker ---

  public tripSystem1CircuitBreaker(
    reason: System1TripReason,
    details: Record<string, unknown> = {},
  ): void {
    // Avoid duplicate alert storms while already tripped on the same fault
    if (this.system1Breaker.tripped && this.system1Breaker.reason === reason) {
      return;
    }

    this.system1Breaker = {
      tripped: true,
      reason,
      trippedAt: Date.now(),
      details,
    };

    // Keep ExitSupervisor active so open positions still execute stop-loss / TP waterfalls
    this.emit('risk:system1_tripped', this.system1Breaker);
  }

  public resetSystem1CircuitBreaker(): void {
    this.system1Breaker = {
      tripped: false,
      reason: null,
      trippedAt: null,
      details: {},
    };
    this.emit('risk:system1_reset');
  }

  public canOpenSystem1Entry(): boolean {
    return !this.system1Breaker.tripped && !this.isDailyDrawdownHalted();
  }

  public getSystem1BreakerState(): System1BreakerState {
    return { ...this.system1Breaker };
  }

  // --- System 2 Circuit Breaker ---

  public tripSystem2CircuitBreaker(
    reason: System2TripReason,
    details: Record<string, unknown> = {},
  ): void {
    if (this.system2Breaker.tripped && this.system2Breaker.reason === reason) {
      return;
    }

    this.system2Breaker = {
      tripped: true,
      reason,
      trippedAt: Date.now(),
      details,
    };

    this.emit('risk:system2_tripped', this.system2Breaker);
  }

  public resetSystem2CircuitBreaker(): void {
    if (!this.system2Breaker.tripped) return;

    this.system2Breaker = {
      tripped: false,
      reason: null,
      trippedAt: null,
      details: {},
    };

    this.emit('risk:system2_reset');
  }

  public canOpenSystem2Entry(): boolean {
    return !this.system2Breaker.tripped && !this.isDailyDrawdownHalted();
  }

  public getSystem2BreakerState(): System2BreakerState {
    return { ...this.system2Breaker };
  }

    const s1CooldownMs = cooldowns.system1Ms ?? 60_000;
    const s2OomCooldownMs = cooldowns.system2OomMs ?? 120_000;

    const onSystem1Tripped = (state: System1BreakerState): void => {
      const reason = state.reason ?? 'UNKNOWN';
      const isStallOrCrash =
        reason === 'SYSTEM1_HEARTBEAT_LOSS' || reason === 'SYSTEM1_WORKER_EXIT';

      notifier.notify({

        title: isStallOrCrash
          ? 'System 1 Fast Reflex Stall / Exit'
          : 'System 1 Latency Budget Degraded',
        summary:
          'UnifiedRiskGovernor halted new System 1 entries. ExitSupervisor remains active for open positions.',
        cooldownMs: s1CooldownMs,
        fields: {
          Reason: reason,
          HeartbeatAgeMs:
            typeof state.details.heartbeatAgeMs === 'number'
              ? state.details.heartbeatAgeMs.toFixed(1)
              : undefined,
          LastSeq: state.details.lastSeq as string | undefined,
          EldP99Ms:
            typeof state.details.eldP99Ms === 'number'
              ? state.details.eldP99Ms.toFixed(2)
              : undefined,
          ExitCode: state.details.exitCode as number | undefined,
        },
      });
    };

    const onSystem1Reset = (): void => {
      notifier.clearCooldown('RISK_S1_SYSTEM1_HEARTBEAT_LOSS');
      notifier.clearCooldown('RISK_S1_SYSTEM1_LATENCY_DEGRADED');
      notifier.clearCooldown('RISK_S1_SYSTEM1_WORKER_EXIT');

      notifier.notify({
        dedupeKey: 'RISK_S1_RECOVERY',
        severity: 'RECOVERY',
        title: 'System 1 Circuit Breaker Reset',
        summary: 'Fast Reflex Engine entries re-enabled.',
        bypassCooldown: true,
      });
    };

    const onSystem2Tripped = (state: System2BreakerState): void => {
      const reason = state.reason ?? 'UNKNOWN';
      // Exhaustion is handled with dedicated supervisor context by NexusFabricRouter
      if (reason === 'SYSTEM2_MAX_RESPAWNS_EXCEEDED') {
        return;
      }

      notifier.notify({
        dedupeKey: `RISK_S2_${reason}`,
        severity: 'WARNING',
        title: 'System 2 Deep Confluence Worker Crash',
        summary:
          'System 2 entries paused while NexusFabricRouter schedules an exponential backoff respawn.',
        cooldownMs: s2OomCooldownMs,
        fields: {
          Reason: reason,
          Generation: state.details.generation as number | undefined,
          Attempt: state.details.attempt as number | undefined,
          ErrorCode: state.details.code as string | undefined,
        },
      });
    };

    const onSystem2Reset = (): void => {
      notifier.clearCooldown('RISK_S2_SYSTEM2_OOM_CRASH');
      notifier.clearCooldown('RISK_S2_SYSTEM2_WORKER_EXIT');
      notifier.clearCooldown('NEXUS_S2_RESPAWN_EXHAUSTED');
    };

    this.on('risk:system1_tripped', onSystem1Tripped);
    this.on('risk:system1_reset', onSystem1Reset);
    this.on('risk:system2_tripped', onSystem2Tripped);
    this.on('risk:system2_reset', onSystem2Reset);

    return () => {
      this.off('risk:system1_tripped', onSystem1Tripped);
      this.off('risk:system1_reset', onSystem1Reset);
      this.off('risk:system2_tripped', onSystem2Tripped);
      this.off('risk:system2_reset', onSystem2Reset);
    };
  }
}

  AlertSeverity,
  { emoji: string; discordColor: number }
> = {

  WARNING: { emoji: '⚠️', discordColor: 0xf59e0b },  // Amber-500

function escapeHtml(raw: string): string {

    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')

export class OpsWebhookNotifier {
  private readonly telegramBotToken?: string;
  private readonly telegramChatId?: string;
  private readonly discordWebhookUrl?: string;
  private readonly defaultCooldownMs: number;
  private readonly requestTimeoutMs: number;
  private readonly environmentLabel: string;
  private readonly lastSentByKey = new Map<string, number>();
  private readonly inFlight = new Set<Promise<void>>();

  constructor(config: OpsWebhookNotifierConfig) {
    this.telegramBotToken = config.telegramBotToken?.trim() || undefined;
    this.telegramChatId = config.telegramChatId?.trim() || undefined;
    this.discordWebhookUrl = config.discordWebhookUrl?.trim() || undefined;
    this.defaultCooldownMs = config.defaultCooldownMs ?? 60_000;
    this.requestTimeoutMs = config.requestTimeoutMs ?? 4_000;
    this.environmentLabel = config.environmentLabel ?? 'MIA-5.0';
  }

  public isConfigured(): boolean {
    const hasTelegram = Boolean(this.telegramBotToken && this.telegramChatId);
    const hasDiscord = Boolean(this.discordWebhookUrl);
    return hasTelegram || hasDiscord;
  }

  /**
   * Fire-and-forget dispatch with cooldown deduplication.
   * Returns true if the alert was accepted for delivery, false if suppressed by cooldown.
   */
  public notify(alert: OpsAlertPayload): boolean {
    const now = Date.now();
    const cooldownMs = alert.cooldownMs ?? this.defaultCooldownMs;

    if (!alert.bypassCooldown && cooldownMs > 0) {
      const lastSentAt = this.lastSentByKey.get(alert.dedupeKey);
      if (lastSentAt !== undefined && now - lastSentAt < cooldownMs) {
        return false;
      }
    }

    this.lastSentByKey.set(alert.dedupeKey, now);

    if (!this.isConfigured()) {
      return true;
    }

    const task = this.dispatchAll(alert, now).finally(() => {
      this.inFlight.delete(task);
    });
    this.inFlight.add(task);

    return true;
  }

  /**
   * Clears cooldown for a fault key upon recovery so a subsequent trip fires immediately.
   */
  public clearCooldown(dedupeKey: string): void {
    this.lastSentByKey.delete(dedupeKey);
  }

   * Used during graceful shutdown or integration tests to await pending HTTP calls.
   */
  public async flush(): Promise<void> {
    await Promise.allSettled([...this.inFlight]);
  }

  private async dispatchAll(
    alert: OpsAlertPayload,
    timestampMs: number,
  ): Promise<void> {
    const deliveries: Promise<void>[] = [];

    if (this.telegramBotToken && this.telegramChatId) {
      deliveries.push(this.sendTelegram(alert, timestampMs));

    if (this.discordWebhookUrl) {
      deliveries.push(this.sendDiscord(alert, timestampMs));
    }

      escapeHtml(alert.summary),
      ...(fieldLines.length > 0 ? ['', ...fieldLines] : []),
      '',

    const timer = setTimeout(() => {
      nexus.off('reflex:telemetry', onTelemetry);
      reject(new Error('Timed out waiting for initial HEALTHY reflex heartbeat'));
    }, timeoutMs);

    const onTelemetry = (dto: { healthy: boolean; stateLabel: string }) => {
      if (dto.healthy && dto.stateLabel === 'HEALTHY') {
        clearTimeout(timer);
        nexus.off('reflex:telemetry', onTelemetry);
        resolve();
      }
    };

    nexus.on('reflex:telemetry', onTelemetry);
  });
}

function waitForDeepReady(
  nexus: NexusFabricRouter,
  timeoutMs = 1_500,
): Promise<DeepRecoveredEvent | void> {
  if (nexus.isDeepWorkerReady()) {
    return Promise.resolve();
  }

  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => {
      nexus.off('nexus:deep_recovered', onRecovered);
      reject(new Error('Timed out waiting for deepConfluenceWorker READY'));
    }, timeoutMs);

    const onRecovered = (evt: DeepRecoveredEvent) => {
      clearTimeout(timer);
      resolve(evt);
    };

    nexus.once('nexus:deep_recovered', onRecovered);
  });
}
