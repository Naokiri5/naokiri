function generateTestSolanaKeypair() {​
  const { publicKey, privateKey } = crypto.generateKeyPairSync('ed25519');​
  const privDer = privateKey.export({ format: 'der', type: 'pkcs8' });​
  const pubDer = publicKey.export({ format: 'der', type: 'spki' });​
​
  const seed32 = new Uint8Array(privDer.subarray(privDer.length - 32));​
  const pub32 = new Uint8Array(pubDer.subarray(pubDer.length - 32));​
​
  const full64 = new Uint8Array(64);​
  full64.set(seed32, 0);​
  full64.set(pub32, 32);​
​
  return {​
    secretKeyBase58: encodeBase58(full64),​
    publicKeyBytes: pub32,​
    publicKeyBase58: encodeBase58(pub32),​
    publicKeyObj: publicKey,​
  };​
}​

function buildMockSplMintBase64(options: {​
  mintAuthorityActive: boolean;​
  freezeAuthorityActive: boolean;​
}): string {​
    const buf = Buffer.alloc(82, 0);​
    // [0..4] mintAuthorityOption: 0 = revoked, 1 = active​
  buf.writeUInt32LE(options.mintAuthorityActive ? 1 : 0, 0);​
  // [45] isInitialized = 1​
  buf.writeUInt8(1, 45);​
  // [46..50] freezeAuthorityOption: 0 = revoked, 1 = active​
