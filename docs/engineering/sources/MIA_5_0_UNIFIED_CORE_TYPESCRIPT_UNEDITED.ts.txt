/**​
 * @id MIA-CORE-01​
 * @description Single typed event bus enforcing unidirectional DAG permissions and 5s TCA
markout sampling.​
 */​
export class NexusFabricRouter extends EventEmitter {​
    private readonly ports = new Map<string, RegisteredPort>();​
    private readonly latestFrames = new Map<string, MarketFrame>();​
    private readonly latestRegimes = new Map<string, RegimeSnapshot>();​
    private readonly pendingMarkouts = new Map<string, NodeJS.Timeout>();​
    private isHalted = false;​
​
    public registerWorkerPort(name: string, role: WorkerRole, port: MessagePort | Worker): void
{​
     if (this.ports.has(name)) {​
      throw new Error(`[MIA-CORE-01] Duplicate worker registration: "${name}"`);​
     }​
​
     this.ports.set(name, { name, role, port });​
​
     port.on('message', (envelope: { topic?: keyof FabricTopicMap; payload?: unknown }) => {​
       if (!envelope || typeof envelope.topic !== 'string') return;​
       this.publishFromRole(role, name, envelope.topic, envelope.payload as never);​
     });​
​
      port.on('error', (error) => {​
        this.emitTyped('fabric:error', { source: name, error, tsMs: Date.now() });​
      });​
    }​
​
    public publishFromRole<K extends keyof FabricTopicMap>(​
      role: WorkerRole,​
      sourceName: string,​
      topic: K,​
      payload: Parameters<FabricTopicMap[K]>[0]​
    ): boolean {​
      const allowed = ALLOWED_PUBLISHERS[topic];​
      if (!allowed.includes(role)) {​
        this.emitTyped('fabric:error', {​
          source: sourceName,​
          error: new Error(​
       `[MIA-CORE-01] UNIDIRECTIONAL_VIOLATION: Role "${role}" (${sourceName}) cannot
publish "${topic}"`​

          ),​
          tsMs: Date.now(),​
         });​
         return false;​
     }​
​
     if (this.isHalted && topic === 'signal:candidate') {​
         return false;​
     }​
​
     if (topic === 'market:frame') {​
         const frame = payload as MarketFrame;​
         this.latestFrames.set(frame.symbol, frame);​
     } else if (topic === 'regime:update') {​
       const regime = payload as RegimeSnapshot;​
       this.latestRegimes.set(regime.symbol, regime);​
     } else if (topic === 'oms:fill') {​
       this.schedulePostFillMarkout(payload as FillEvent);​
     }​
​
     this.emitTyped(topic, payload);​
     this.fanOutToPorts(topic, payload, sourceName);​
     return true;​
    }​
​
    public emitTyped<K extends keyof FabricTopicMap>(​
      topic: K,​
      payload: Parameters<FabricTopicMap[K]>[0]​
    ): void {​
      super.emit(topic, payload);​
    }​
​
    public onTyped<K extends keyof FabricTopicMap>(topic: K, listener: FabricTopicMap[K]):
this {​
     super.on(topic, listener);​
     return this;​
    }​
​
    public getLatestFrame(symbol: string): MarketFrame | undefined {​
     return this.latestFrames.get(symbol);​
    }​
​
    public getLatestRegime(symbol: string): RegimeSnapshot | undefined {​

     return this.latestRegimes.get(symbol);​
    }​
​
    public setGlobalHalt(halted: boolean): void {​
     this.isHalted = halted;​
    }​
​
    private fanOutToPorts<K extends keyof FabricTopicMap>(​
    topic: K,​
    payload: Parameters<FabricTopicMap[K]>[0],​
    originName: string​
  ): void {​
    for (const [name, entry] of this.ports.entries()) {​
      if (name === originName) continue;​
      if (entry.role === 'STRATEGY' && topic !== 'market:frame' && topic !== 'regime:update')
continue;​
      if (entry.role === 'OMS' && topic !== 'order:approved') continue;​
      entry.port.postMessage({ topic, payload });​
    }​
  }​
​
  private schedulePostFillMarkout(fill: FillEvent): void {​
    const timer = setTimeout(() => {​
      this.pendingMarkouts.delete(fill.clientOrderId);​
      const frame = this.latestFrames.get(fill.symbol);​
      if (!frame) return;​
​
      const mid5s = (frame.bestBid + frame.bestAsk) / 2;​
      const direction = fill.side === 'BUY' ? 1 : -1;​
      const markoutBps = ((mid5s - fill.avgFillPrice) / fill.avgFillPrice) * 10_000 * direction;​
​
      const engine: EngineSource = fill.clientOrderId.startsWith('EX_')​
        ? 'EXIT_SUPERVISOR'​
        : fill.clientOrderId.startsWith('FR_')​
          ? 'FAST_REFLEX'​
          : 'DEEP_CONFLUENCE';​
​
      const sample: MarkoutSample = {​
        clientOrderId: fill.clientOrderId,​
        symbol: fill.symbol,​
        engine,​
        side: fill.side,​
        fillPrice: fill.avgFillPrice,​

          midPrice5sLater: mid5s,​
         markoutBps,​
         tsMs: Date.now(),​
       };​
​
       this.publishFromRole('LEDGER', 'MarkoutSampler', 'tca:markout', sample);​
     }, 5_000);​
​
      timer.unref();​
      this.pendingMarkouts.set(fill.clientOrderId, timer);​
    }​
​
    public shutdown(): void {​
     for (const timer of this.pendingMarkouts.values()) {​
       clearTimeout(timer);​
     }​
     this.pendingMarkouts.clear();​
  }​
}​

        qty = excluded.qty,​
        entry_price = excluded.entry_price,​
        updated_at_ms = excluded.updated_at_ms`​
     )​
     .run(​
       reconciled.positionId,​
       reconciled.symbol,​
       reconciled.side,​
       reconciled.qty,​
       reconciled.entryPrice,​
       reconciled.stopLossPrice,​
       reconciled.realizedPnlUsd,​
       reconciled.engine,​
       reconciled.openedAtMs,​
       reconciled.updatedAtMs​
     );​
​
    const { engine: _e, openedAtMs: _o, ...canonical } = reconciled;​
    this.router.publishFromRole('LEDGER', 'CanonicalPositionLedger', 'ledger:position',
canonical);​
    }​
  }​
​
  private updateUnrealizedPnl(frame: MarketFrame): void {​
    const pos = this.positionsBySymbol.get(frame.symbol);​

    if (!pos || pos.qty <= 0) return;​
​
    const markPrice = pos.side === 'BUY' ? frame.bestBid : frame.bestAsk;​
    if (markPrice <= 0) return;​
​
    const direction = pos.side === 'BUY' ? 1 : -1;​
    const unrealized = (markPrice - pos.entryPrice) * pos.qty * direction;​
​
    if (Math.abs(unrealized - pos.unrealizedPnlUsd) >= 0.05) {​
      pos.unrealizedPnlUsd = unrealized;​
      pos.updatedAtMs = Date.now();​
      const { engine: _e, openedAtMs: _o, ...canonical } = pos;​
      this.router.publishFromRole('LEDGER', 'CanonicalPositionLedger', 'ledger:position',
canonical);​
    }​
  }​
​
  public close(): void {​
    this.db.close();​
  }​
}​

  },​

/**​
 * @id MIA-CORE-03​
 * @description Sole authority permitted to mint ApprovedOrderIntent; enforces drawdown, VPIN,
correlation, and Kelly sizing.​

*/​
export class UnifiedRiskGovernor {​
    private equityUsd: number;​
    private dayStartEquityUsd: number;​
    private realizedDailyPnlUsd = 0;​
    private isDrawdownHalted = false;​
​
    private readonly openPositions = new Map<string, CanonicalPosition>();​
    private readonly markoutHistory = new Map<string, MarkoutSample[]>();​
    private readonly toxicityCooldownUntil = new Map<string, number>();​
​
    constructor(​
      private readonly router: NexusFabricRouter,​
      private readonly config: RiskGovernorConfig,​
      private readonly macroGate: MacroCalendarGate,​
      private readonly expectancyProvider: StrategyExpectancyProvider​
    ) {​
     this.equityUsd = config.initialEquityUsd;​
     this.dayStartEquityUsd = config.initialEquityUsd;​
     this.bindRouterSubscriptions();​
    }​
​
    private bindRouterSubscriptions(): void {​
     this.router.onTyped('signal:candidate', (signal) => this.evaluateCandidate(signal));​
     this.router.onTyped('exit:intent', (exitIntent) => this.approveExitIntent(exitIntent));​
     this.router.onTyped('ledger:position', (pos) => this.onPositionUpdate(pos));​
     this.router.onTyped('tca:markout', (sample) => this.onMarkoutSample(sample));​
    }​
​
    public reconcileStartupState(equityUsd: number, realizedDailyPnlUsd: number): void {​
     this.equityUsd = equityUsd;​
     this.dayStartEquityUsd = equityUsd - realizedDailyPnlUsd;​
     this.realizedDailyPnlUsd = realizedDailyPnlUsd;​
     this.checkDrawdownBreakers();​
    }​
​
    public resetDailyAnchor(): void {​
     const unrealized = this.computeTotalUnrealizedPnl();​
     this.dayStartEquityUsd = this.equityUsd + unrealized;​
     this.realizedDailyPnlUsd = 0;​
     this.isDrawdownHalted = false;​
     this.router.setGlobalHalt(false);​
    }​

​
    private onPositionUpdate(pos: CanonicalPosition): void {​
     if (pos.qty <= 0) {​
       const prev = this.openPositions.get(pos.positionId);​
       if (prev) {​
          const deltaRealized = pos.realizedPnlUsd - prev.realizedPnlUsd;​
          this.realizedDailyPnlUsd += deltaRealized;​
          this.equityUsd += deltaRealized;​
       }​
       this.openPositions.delete(pos.positionId);​
      } else {​
        this.openPositions.set(pos.positionId, pos);​
      }​
      this.checkDrawdownBreakers();​
    }​
​
    private onMarkoutSample(sample: MarkoutSample): void {​
     const key = `${sample.engine}:${sample.symbol}`;​
     const list = this.markoutHistory.get(key) ?? [];​
     list.push(sample);​
     if (list.length > 10) list.shift();​
     this.markoutHistory.set(key, list);​
​
     if (list.length >= 5) {​
       const avgMarkoutBps = list.reduce((acc, s) => acc + s.markoutBps, 0) / list.length;​
       if (avgMarkoutBps <= this.config.markoutToxicityThresholdBps) {​
          this.toxicityCooldownUntil.set(key, Date.now() + this.config.toxicityCooldownMs);​
        }​
      }​
    }​
​
    private evaluateCandidate(signal: CandidateSignal): void {​
     const nowMs = Date.now();​
​
     if (this.isDrawdownHalted) {​
       return this.veto(signal, 'CIRCUIT_BREAKER_DRAWDOWN', 'Daily drawdown limit breached');​
     }​
​
     const macroStatus = this.macroGate.isEventVetoActive(signal.symbol, nowMs);​
     if (macroStatus.blocked) {​
       return this.veto(signal, 'MACRO_CALENDAR_VETO', macroStatus.reason ?? 'High-impact
macro event');​
     }​

​
    const frame = this.router.getLatestFrame(signal.symbol);​
    if (!frame || frame.bestAsk <= frame.bestBid) {​
     return this.veto(signal, 'SPREAD_OR_DEPTH_GUARD', 'Missing or crossed L1 market
frame');​
    }​
​
    const spread = frame.bestAsk - frame.bestBid;​
    if (frame.atr14 > 0 && spread / frame.atr14 > this.config.maxSpreadToAtrRatio) {​
     return this.veto(​
         signal,​
         'SPREAD_OR_DEPTH_GUARD',​
         `Spread/ATR ratio ${(spread / frame.atr14).toFixed(3)} exceeds cap`​
      );​
    }​
​
    const toxicityKey = `${signal.engine}:${signal.symbol}`;​
    const cooldownUntil = this.toxicityCooldownUntil.get(toxicityKey) ?? 0;​
    if (nowMs < cooldownUntil) {​
     return this.veto(​
         signal,​
         'TOXICITY_VPIN_HALT',​
         `Adverse selection cooldown active until ${new Date(cooldownUntil).toISOString()}`​
      );​
    }​
​
    if (signal.engine === 'FAST_REFLEX' && frame.vpin >= this.config.vpinHaltThreshold) {​
     return this.veto(​
         signal,​
         'TOXICITY_VPIN_HALT',​
         `Live VPIN ${frame.vpin.toFixed(2)} >= ${this.config.vpinHaltThreshold}`​
      );​
    }​
​
    const activeRegime: MarketRegime =​
     this.router.getLatestRegime(signal.symbol)?.regime ?? 'LOW_VOL_CHOP';​
    const regimeMultiplier = REGIME_SIZE_MULTIPLIERS[signal.engine][activeRegime];​
​
    if (regimeMultiplier <= 0) {​
     return this.veto(signal, 'REGIME_BLOCKED', `Engine ${signal.engine} disabled in
${activeRegime}`);​
    }​
​

    const stopDistance = Math.abs(signal.entryPrice - signal.stopLossPrice);​
    if (stopDistance <= 0) {​
     return this.veto(signal, 'MIN_NOTIONAL_OR_ZERO_SIZE', 'Invalid zero stop-loss distance');​
    }​
​
    const kellyScale = this.computeFractionalKellyScale(signal.engine, activeRegime);​
    const effectiveRiskPct =​
     this.config.baseRiskPerTradePct * regimeMultiplier * kellyScale * signal.confidence;​
    const riskUsd = this.equityUsd * effectiveRiskPct;​
​
    let qty = riskUsd / stopDistance;​
    let notionalUsd = qty * signal.entryPrice;​
​
    const relevantDepth = signal.side === 'BUY' ? frame.askDepthTop3 :
frame.bidDepthTop3;​
   const maxBookAllowedNotional =​
     (relevantDepth * signal.entryPrice) / this.config.minBookDepthMultiple;​
   if (notionalUsd > maxBookAllowedNotional) {​
     notionalUsd = maxBookAllowedNotional;​
     qty = notionalUsd / signal.entryPrice;​
   }​
​
   const remainingBucketCapacityUsd = this.computeRemainingBucketCapacity(​
     signal.symbol,​
     signal.side​
   );​
   const remainingTotalCapacityUsd =​
     this.equityUsd * this.config.maxTotalNotionalMultiplier -
this.computeTotalGrossNotional();​
​
   const allowedNotionalUsd = Math.min(​
     notionalUsd,​
     remainingBucketCapacityUsd,​
     remainingTotalCapacityUsd​
   );​
​
   if (allowedNotionalUsd < 15) {​
     return this.veto(​
       signal,​
       'CORRELATED_DELTA_CAP',​
         `Remaining correlated/portfolio capacity ($${allowedNotionalUsd.toFixed(2)}) below minimum`​
      );​
    }​

​
     const finalQty = allowedNotionalUsd / signal.entryPrice;​
     const prefix = signal.engine === 'FAST_REFLEX' ? 'FR' : 'DC';​
​
     const approved: ApprovedOrderIntent = {​
         clientOrderId: `${prefix}_${signal.symbol}_${randomUUID().slice(0, 8)}`,​
         signalId: signal.signalId,​
         engine: signal.engine,​
         symbol: signal.symbol,​
         side: signal.side,​
         orderType: signal.engine === 'FAST_REFLEX' ? 'LIMIT_IOC' : 'LIMIT_POST_ONLY',​
         limitPrice: signal.entryPrice,​
         qty: Number(finalQty.toFixed(6)),​
         notionalUsd: allowedNotionalUsd,​
         stopLossPrice: signal.stopLossPrice,​
         takeProfitPrice: signal.takeProfitPrice,​
         regimeAtEntry: activeRegime,​
         riskUsd: finalQty * stopDistance,​
         approvedAtMs: nowMs,​
     };​
​
     this.router.publishFromRole('RISK', 'UnifiedRiskGovernor', 'order:approved', approved);​
    }​
​
    private approveExitIntent(exit: ExitOrderIntent): void {​
     const frame = this.router.getLatestFrame(exit.symbol);​
     const refPrice = frame ? (exit.side === 'SELL' ? frame.bestBid : frame.bestAsk) : 0;​
​
     const approvedExit: ApprovedOrderIntent = {​
         clientOrderId: `EX_S${exit.stage}_${exit.symbol}_${randomUUID().slice(0, 8)}`,​
         signalId: exit.intentId,​
         engine: 'EXIT_SUPERVISOR',​
         symbol: exit.symbol,​
         side: exit.side,​
         orderType: exit.urgentMarket ? 'MARKET' : 'LIMIT_IOC',​
         limitPrice: refPrice,​
         qty: Number(exit.qty.toFixed(6)),​
         notionalUsd: exit.qty * refPrice,​
         stopLossPrice: 0,​
         takeProfitPrice: 0,​
         regimeAtEntry: this.router.getLatestRegime(exit.symbol)?.regime ?? 'LOW_VOL_CHOP',​
         riskUsd: 0,​
         approvedAtMs: Date.now(),​

     };​
​
     this.router.publishFromRole('RISK', 'UnifiedRiskGovernor', 'order:approved', approvedExit);​
    }​
​
    private computeFractionalKellyScale(engine: EngineSource, regime: MarketRegime):
number {​
     const stats = this.expectancyProvider.getRollingExpectancy(engine, regime);​
     if (stats.sampleCount < 15 || stats.payoffRatio <= 0) return 0.75;​
     const kellyFraction = stats.winRate - (1 - stats.winRate) / stats.payoffRatio;​
     const quarterKelly = kellyFraction * 0.25;​
     return Math.max(0.35, Math.min(1.35, quarterKelly / 0.02));​
    }​
​
    private computeRemainingBucketCapacity(symbol: string, side: TradeSide): number {​
     const targetBucket = CORRELATION_BUCKETS[symbol] ?? 'OTHER_ALT';​
     const maxBucketDeltaUsd = this.equityUsd * this.config.maxCorrelatedDeltaMultiplier;​
​
     let signedBucketDeltaUsd = 0;​
     for (const pos of this.openPositions.values()) {​
         const posBucket = CORRELATION_BUCKETS[pos.symbol] ?? 'OTHER_ALT';​
         if (posBucket !== targetBucket) continue;​
         const sign = pos.side === 'BUY' ? 1 : -1;​
       signedBucketDeltaUsd += sign * pos.qty * pos.entryPrice;​
     }​
​
     const incomingSign = side === 'BUY' ? 1 : -1;​
     const projectedDirectionalExposure = signedBucketDeltaUsd * incomingSign;​
     return Math.max(0, maxBucketDeltaUsd - projectedDirectionalExposure);​
    }​
​
    private computeTotalGrossNotional(): number {​
     let gross = 0;​
     for (const pos of this.openPositions.values()) {​
        gross += pos.qty * pos.entryPrice;​
      }​
      return gross;​
    }​
​
    private computeTotalUnrealizedPnl(): number {​
     let unrealized = 0;​
     for (const pos of this.openPositions.values()) {​
         unrealized += pos.unrealizedPnlUsd;​

     }​
     return unrealized;​
    }​
​
    private checkDrawdownBreakers(): void {​
     const totalDailyPnl = this.realizedDailyPnlUsd + this.computeTotalUnrealizedPnl();​
     const drawdownPct = -totalDailyPnl / Math.max(1, this.dayStartEquityUsd);​
​
     if (drawdownPct >= this.config.maxDailyDrawdownPct && !this.isDrawdownHalted) {​
         this.isDrawdownHalted = true;​
         this.router.setGlobalHalt(true);​
     }​
​
     if (drawdownPct >= this.config.hardFlattenDrawdownPct) {​
         for (const pos of this.openPositions.values()) {​
          this.approveExitIntent({​
           intentId: `EMERGENCY_DD_${pos.positionId}`,​
           positionId: pos.positionId,​
           symbol: pos.symbol,​
           side: pos.side === 'BUY' ? 'SELL' : 'BUY',​
           qty: pos.qty,​
           stage: 1,​
           reason: `Hard drawdown limit ${(drawdownPct * 100).toFixed(2)}% breached`,​
           urgentMarket: true,​
           tsMs: Date.now(),​
          });​
        }​
      }​
    }​
​
    private veto(​
      signal: CandidateSignal,​
      reasonCode: RiskVetoEvent['reasonCode'],​
      detail: string​
    ): void {​
      this.router.publishFromRole('RISK', 'UnifiedRiskGovernor', 'risk:veto', {​
        signalId: signal.signalId,​
        engine: signal.engine,​
        symbol: signal.symbol,​
        reasonCode,​
        detail,​
        tsMs: Date.now(),​
      });​

  }​
}​

export interface ExitSupervisorConfig {​
    feeBufferBps: number;​
  inFlightTimeoutMs: number;​
  ofiExhaustionThreshold: number;​
  vpinEmergencyExitThreshold: number;​
}​
​
interface RegimeExitProfile {​
  partialTpR: number;​
  partialCloseFraction: number;​
  atrTrailMultiplier: number;​
  maxHoldMs: number;​
}​
​
const REGIME_EXIT_PROFILES: Record<MarketRegime, RegimeExitProfile> = {​
  LOW_VOL_CHOP: { partialTpR: 0.85, partialCloseFraction: 0.6, atrTrailMultiplier: 1.8,
maxHoldMs: 12 * 60_000 },​
  TRENDING: { partialTpR: 1.5, partialCloseFraction: 0.4, atrTrailMultiplier: 2.6, maxHoldMs: 45 *
60_000 },​
  HIGH_VOL_EXPANSION: { partialTpR: 1.2, partialCloseFraction: 0.5, atrTrailMultiplier: 2.2,
maxHoldMs: 25 * 60_000 },​
  LIQUIDATION_CASCADE: { partialTpR: 0.75, partialCloseFraction: 0.75, atrTrailMultiplier: 1.2,
maxHoldMs: 5 * 60_000 },​
};​
​
interface ManagedPositionState {​
  position: CanonicalPosition;​
  initialStopPrice: number;​
  initialRiskPerUnit: number;​
  currentStopPrice: number;​
  highestFavoredPrice: number;​
  openedAtMs: number;​
  stage2PartialTaken: boolean;​
  breakevenLocked: boolean;​

  inFlightExitUntilMs: number;​
}​
​
/**​
 * @id MIA-CORE-04​
 * @description Non-greedy 5-stage exit waterfall driven exclusively by CanonicalPositionLedger
updates.​
 */​
export class ExitSupervisor {​
  private readonly managed = new Map<string, ManagedPositionState>();​
​
    constructor(​
      private readonly router: NexusFabricRouter,​
      private readonly config: ExitSupervisorConfig = {​
        feeBufferBps: 12,​
        inFlightTimeoutMs: 3_000,​
        ofiExhaustionThreshold: 0.55,​
        vpinEmergencyExitThreshold: 0.82,​
      }​
    ) {​
     this.router.onTyped('ledger:position', (pos) => this.syncPositionFromLedger(pos));​
     this.router.onTyped('oms:fill', (fill) => this.onFillConfirmed(fill));​
     this.router.onTyped('market:frame', (frame) => this.evaluateMarketFrame(frame));​
    }​
​
    private syncPositionFromLedger(pos: CanonicalPosition): void {​
     if (pos.qty <= 0) {​
         this.managed.delete(pos.positionId);​
         return;​
     }​
​
     const existing = this.managed.get(pos.positionId);​
     if (!existing) {​
         const initialRiskPerUnit = Math.abs(pos.entryPrice - pos.stopLossPrice);​
         this.managed.set(pos.positionId, {​
          position: pos,​
          initialStopPrice: pos.stopLossPrice,​
          initialRiskPerUnit: initialRiskPerUnit > 0 ? initialRiskPerUnit : pos.entryPrice * 0.005,​
          currentStopPrice: pos.stopLossPrice,​
          highestFavoredPrice: pos.entryPrice,​
          openedAtMs: pos.updatedAtMs,​
          stage2PartialTaken: false,​
          breakevenLocked: false,​
          inFlightExitUntilMs: 0,​

      });​
      return;​
     }​
​
      existing.position = pos;​
    }​
​
    private onFillConfirmed(fill: FillEvent): void {​
     const state = this.managed.get(fill.positionId);​
     if (!state) return;​
​
     state.inFlightExitUntilMs = 0;​
​
     if (fill.clientOrderId.startsWith('EX_S2_')) {​
        state.stage2PartialTaken = true;​
        state.breakevenLocked = true;​
        const feeOffset = state.position.entryPrice * (this.config.feeBufferBps / 10_000);​
        state.currentStopPrice =​
         state.position.side === 'BUY'​
           ? Math.max(state.currentStopPrice, state.position.entryPrice + feeOffset)​
           : Math.min(state.currentStopPrice, state.position.entryPrice - feeOffset);​
      }​
    }​
​
    private evaluateMarketFrame(frame: MarketFrame): void {​
     const nowMs = Date.now();​
     const activeRegime: MarketRegime =​
      this.router.getLatestRegime(frame.symbol)?.regime ?? 'LOW_VOL_CHOP';​
     const profile = REGIME_EXIT_PROFILES[activeRegime];​
​
     for (const state of this.managed.values()) {​
      if (state.position.symbol !== frame.symbol || nowMs < state.inFlightExitUntilMs)
continue;​
​
      const { position } = state;​
      const isLong = position.side === 'BUY';​
      const exitSide: TradeSide = isLong ? 'SELL' : 'BUY';​
      const markPrice = isLong ? frame.bestBid : frame.bestAsk;​
      if (markPrice <= 0) continue;​
​
      if (isLong && markPrice > state.highestFavoredPrice) state.highestFavoredPrice =
markPrice;​
   else if (!isLong && markPrice < state.highestFavoredPrice) state.highestFavoredPrice =

markPrice;​
​
    const favorableMove = isLong ? markPrice - position.entryPrice : position.entryPrice -
markPrice;​
    const currentR = favorableMove / state.initialRiskPerUnit;​
​
    // Stage 1: Hard Stop / Breakeven Stop​
    const stopBreached = isLong​
      ? markPrice <= state.currentStopPrice​
      : markPrice >= state.currentStopPrice;​
    if (stopBreached) {​
      this.emitExitIntent(state, {​
        stage: 1,​
        side: exitSide,​
        qty: position.qty,​
        urgentMarket: true,​
        reason: state.breakevenLocked​
          ? `Stage 1 Breakeven Lock hit @ ${state.currentStopPrice.toFixed(4)}`​
          : `Stage 1 Hard Stop hit @ ${state.currentStopPrice.toFixed(4)}`,​
      });​
      continue;​
    }​
​
    // Stage 2: Regime-Scaled Partial TP + Breakeven Lock​
    if (!state.stage2PartialTaken && currentR >= profile.partialTpR) {​
      const partialQty = Number((position.qty * profile.partialCloseFraction).toFixed(6));​
      if (partialQty > 0) {​
        this.emitExitIntent(state, {​
          stage: 2,​
          side: exitSide,​
          qty: partialQty,​
          urgentMarket: false,​
          reason: `Stage 2 Partial TP (${currentR.toFixed(2)}R >= ${profile.partialTpR}R in
${activeRegime})`,​
        });​
        continue;​
      }​
    }​
​
    // Stage 3: Regime-Scaled ATR Chandelier Trailing Stop​
    if (state.stage2PartialTaken && frame.atr14 > 0) {​
      const trailOffset = frame.atr14 * profile.atrTrailMultiplier;​
      const candidateTrailStop = isLong​

      ? state.highestFavoredPrice - trailOffset​
      : state.highestFavoredPrice + trailOffset;​
​
     if (isLong && candidateTrailStop > state.currentStopPrice) {​
       state.currentStopPrice = candidateTrailStop;​
     } else if (!isLong && candidateTrailStop < state.currentStopPrice) {​
       state.currentStopPrice = candidateTrailStop;​
     }​
​
     const trailHit = isLong​
        ? markPrice <= state.currentStopPrice​
        : markPrice >= state.currentStopPrice;​
      if (trailHit) {​
        this.emitExitIntent(state, {​
          stage: 3,​
          side: exitSide,​
          qty: position.qty,​
          urgentMarket: true,​
          reason: `Stage 3 ATR Trail (${profile.atrTrailMultiplier}x ATR) hit @
${state.currentStopPrice.toFixed(4)}`,​
        });​
        continue;​
      }​
    }​
​
    // Stage 4: L2 OFI Exhaustion & VPIN Toxicity Surge Exit​
    if (currentR >= 0.35) {​
      const opposingOfi = isLong​
        ? frame.ofiL5 <= -this.config.ofiExhaustionThreshold​
        : frame.ofiL5 >= this.config.ofiExhaustionThreshold;​
      const extremeToxicity = frame.vpin >= this.config.vpinEmergencyExitThreshold;​
​
      if (opposingOfi || extremeToxicity) {​
        this.emitExitIntent(state, {​
          stage: 4,​
          side: exitSide,​
          qty: position.qty,​
          urgentMarket: extremeToxicity,​
          reason: extremeToxicity​
            ? `Stage 4 VPIN Surge (${frame.vpin.toFixed(2)}) locking +${currentR.toFixed(2)}R`​
            : `Stage 4 L5 OFI Exhaustion (${frame.ofiL5.toFixed(2)}) locking
+${currentR.toFixed(2)}R`,​
        });​

           continue;​
         }​
       }​
​
       // Stage 5: Regime-Conditioned Time Stop​
       const holdDurationMs = nowMs - state.openedAtMs;​
       if (holdDurationMs >= profile.maxHoldMs && currentR < 0.4) {​
         this.emitExitIntent(state, {​
           stage: 5,​
           side: exitSide,​
           qty: position.qty,​
           urgentMarket: false,​
           reason: `Stage 5 Time Stop (${Math.round(holdDurationMs / 60_000)}m in ${activeRegime},
${currentR.toFixed(2)}R)`,​
          });​
        }​
      }​
    }​
​
    private emitExitIntent(​
    state: ManagedPositionState,​
    params: { stage: 1 | 2 | 3 | 4 | 5; side: TradeSide; qty: number; urgentMarket: boolean;
reason: string }​
  ): void {​
    const nowMs = Date.now();​
    state.inFlightExitUntilMs = nowMs + this.config.inFlightTimeoutMs;​
​
    this.router.publishFromRole('LEDGER', 'ExitSupervisor', 'exit:intent', {​
      intentId: `EXIT_${state.position.positionId}_S${params.stage}_${randomUUID().slice(0, 6)}`,​
      positionId: state.position.positionId,​
      symbol: state.position.symbol,​
      side: params.side,​
      qty: params.qty,​
      stage: params.stage,​
      reason: params.reason,​
      urgentMarket: params.urgentMarket,​
      tsMs: nowMs,​
    });​
  }​
}​

/**​
 * @id MIA-CORE-05A​
 * @description L2 depth-aware Paper OMS sharing the identical OrderLifecycleState machine as
LiveOmsAdapter.​
 */​
export class PaperOmsAdapter implements OmsAdapter {​
    private readonly orders = new Map<string, PaperOrder>();​
    private readonly restingPostOnlyOrders = new Map<string, PaperOrder>();​
    private readonly timers = new Set<NodeJS.Timeout>();​
​
    constructor(​

     private readonly router: NexusFabricRouter,​
     private readonly config: PaperOmsConfig = {​
       makerFeeBps: 2.0,​
       takerFeeBps: 5.5,​
       simulatedLatencyMs: 0,​
       postOnlyTimeoutMs: 2_500,​
       partialFillThresholdRatio: 0.35,​
     }​
    ) {​
      this.router.onTyped('order:approved', (intent) => this.handleApprovedOrder(intent));​
     this.router.onTyped('market:frame', (frame) => this.matchRestingPostOnlyOrders(frame));​
    }​
​
    public async start(): Promise<void> {​
     // Ready synchronously​
    }​
​
    public handleApprovedOrder(intent: ApprovedOrderIntent): void {​
     if (this.orders.has(intent.clientOrderId)) return;​
​
     const order: PaperOrder = {​
       intent,​
       state: 'PENDING_NEW',​
       filledQty: 0,​
       createdAtMs: Date.now(),​
     };​
     this.orders.set(intent.clientOrderId, order);​
​
     const execute = () => {​
         const frame = this.router.getLatestFrame(intent.symbol);​
         if (!frame || frame.bestAsk <= frame.bestBid) {​
           order.state = 'REJECTED';​
           return;​
         }​
​
         if (intent.orderType === 'LIMIT_POST_ONLY') {​
          const crossesSpread =​
           (intent.side === 'BUY' && intent.limitPrice >= frame.bestAsk) ||​
           (intent.side === 'SELL' && intent.limitPrice <= frame.bestBid);​
​
          if (crossesSpread) {​
           order.intent = {​
            ...intent,​
            limitPrice: intent.side === 'BUY' ? frame.bestBid : frame.bestAsk,​

             };​
           }​
​
           order.state = 'ACKED';​
           this.restingPostOnlyOrders.set(intent.clientOrderId, order);​
​
           const expiry = setTimeout(() => {​
            this.timers.delete(expiry);​
            const resting = this.restingPostOnlyOrders.get(intent.clientOrderId);​
            if (resting && (resting.state === 'ACKED' || resting.state === 'PARTIAL')) {​
            resting.state = 'CANCELED';​
            this.restingPostOnlyOrders.delete(intent.clientOrderId);​
          }​
        }, this.config.postOnlyTimeoutMs);​
        expiry.unref();​
        this.timers.add(expiry);​
        return;​
      }​
​
      this.executeImmediateOrder(order, frame);​
     };​
​
     if (this.config.simulatedLatencyMs > 0) {​
      const timer = setTimeout(() => {​
           this.timers.delete(timer);​
          execute();​
        }, this.config.simulatedLatencyMs);​
        timer.unref();​
        this.timers.add(timer);​
      } else {​
        execute();​
      }​
    }​
​
    private executeImmediateOrder(order: PaperOrder, frame: MarketFrame): void {​
     const { intent } = order;​
     const isBuy = intent.side === 'BUY';​
     const topPrice = isBuy ? frame.bestAsk : frame.bestBid;​
     const availableDepth = isBuy ? frame.askDepthTop3 : frame.bidDepthTop3;​
​
     if (intent.orderType === 'LIMIT_IOC' && intent.limitPrice > 0) {​
      const outOfBounds = isBuy ? topPrice > intent.limitPrice : topPrice < intent.limitPrice;​
      if (outOfBounds) {​

        order.state = 'CANCELED';​
        return;​
      }​
    }​
​
    const depthUtilization = availableDepth > 0 ? intent.qty / availableDepth : 1;​
    const spread = frame.bestAsk - frame.bestBid;​
    const slippage = spread * Math.min(2.5, depthUtilization * 0.85);​
    const execPrice = isBuy ? topPrice + slippage : Math.max(0.0001, topPrice - slippage);​
    const midPriceAtFill = (frame.bestBid + frame.bestAsk) / 2;​
​
    if (availableDepth > 0 && depthUtilization >= this.config.partialFillThresholdRatio) {​
     const firstSlice = Number((intent.qty * 0.6).toFixed(6));​
     const secondSlice = Number((intent.qty - firstSlice).toFixed(6));​
     this.emitFill(order, firstSlice, execPrice, midPriceAtFill, this.config.takerFeeBps, false);​
     if (secondSlice > 0) {​
      const secondPrice = isBuy​
          ? execPrice + spread * 0.25​
          : Math.max(0.0001, execPrice - spread * 0.25);​
        this.emitFill(order, secondSlice, secondPrice, midPriceAtFill, this.config.takerFeeBps,
true);​
      }​
    } else {​
      this.emitFill(order, intent.qty, execPrice, midPriceAtFill, this.config.takerFeeBps, true);​
    }​
  }​
​
  private matchRestingPostOnlyOrders(frame: MarketFrame): void {​
    for (const [clientOrderId, order] of this.restingPostOnlyOrders.entries()) {​
      if (order.intent.symbol !== frame.symbol) continue;​
      const isBuy = order.intent.side === 'BUY';​
      const touched = isBuy​
        ? frame.bestAsk <= order.intent.limitPrice​
        : frame.bestBid >= order.intent.limitPrice;​
      if (!touched) continue;​
​
      const remainingQty = Number((order.intent.qty - order.filledQty).toFixed(6));​
      if (remainingQty > 0) {​
        const midPriceAtFill = (frame.bestBid + frame.bestAsk) / 2;​
        this.emitFill(​
          order,​
          remainingQty,​
          order.intent.limitPrice,​

            midPriceAtFill,​
            this.config.makerFeeBps,​
            true​
          );​
        }​
        this.restingPostOnlyOrders.delete(clientOrderId);​
      }​
    }​
​
    private emitFill(​
      order: PaperOrder,​
      sliceQty: number,​
      fillPrice: number,​
      midPriceAtFill: number,​
      feeBps: number,​
      isFinalSlice: boolean​
    ): void {​
      order.filledQty = Number((order.filledQty + sliceQty).toFixed(6));​
      order.state = isFinalSlice || order.filledQty >= order.intent.qty - 1e-6 ? 'FILLED' : 'PARTIAL';​
​
     this.router.publishFromRole('OMS', 'PaperOmsAdapter', 'oms:fill', {​
       clientOrderId: order.intent.clientOrderId,​
       positionId: order.intent.signalId,​
       symbol: order.intent.symbol,​
       side: order.intent.side,​
       filledQty: sliceQty,​
       avgFillPrice: fillPrice,​
       feeUsd: (sliceQty * fillPrice * feeBps) / 10_000,​
        midPriceAtFill,​
        tsMs: Date.now(),​
      });​
    }​
​
    public shutdown(): void {​
     for (const timer of this.timers) clearTimeout(timer);​
     this.timers.clear();​
     this.restingPostOnlyOrders.clear();​
  }​
}​

/**​
 * @id MIA-CORE-05B​
 * @description Bybit V5 Live OMS with idempotent orderLinkId, 10s Dead-Man's Switch, and
WS/REST position reconciliation.​
 */​
export class LiveOmsAdapter implements OmsAdapter {​
    private ws: WebSocket | null = null;​
    private reconnectAttempts = 0;​
    private isShuttingDown = false;​
    private readonly orders = new Map<string, TrackedLiveOrder>();​
    private deadManTimer: NodeJS.Timeout | null = null;​
    private reconcileTimer: NodeJS.Timeout | null = null;​
    private pingTimer: NodeJS.Timeout | null = null;​
​
    constructor(​
      private readonly router: NexusFabricRouter,​
      private readonly ledger: CanonicalPositionLedger,​
      private readonly config: LiveOmsConfig​
    ) {​
     this.router.onTyped('order:approved', (intent) => {​
       void this.submitApprovedOrder(intent);​
      });​
    }​
​
    public async start(): Promise<void> {​
     this.isShuttingDown = false;​
     await this.reconcilePositionsWithExchange();​

     this.connectPrivateWebSocket();​
​
     const dmsIntervalMs = Math.max(2_000, ((this.config.deadManTimeoutSec ?? 10) *
1_000) / 2.5);​
     this.deadManTimer = setInterval(() => void this.refreshDeadMansSwitch(), dmsIntervalMs);​
     this.deadManTimer.unref();​
​
     this.reconcileTimer = setInterval(​
       () => void this.reconcilePositionsWithExchange(),​
         this.config.reconcileIntervalMs ?? 15_000​
     );​
     this.reconcileTimer.unref();​
    }​
​
    public async submitApprovedOrder(intent: ApprovedOrderIntent): Promise<void> {​
     if (this.orders.has(intent.clientOrderId)) return;​
​
     const tracked: TrackedLiveOrder = {​
       intent,​
       state: 'PENDING_NEW',​
       cumFilledQty: 0,​
       cumFeeUsd: 0,​
       updatedAtMs: Date.now(),​
     };​
     this.orders.set(intent.clientOrderId, tracked);​
​
     const isMarket = intent.orderType === 'MARKET';​
     const body: Record<string, unknown> = {​
         category: 'linear',​
         symbol: intent.symbol,​
         side: intent.side === 'BUY' ? 'Buy' : 'Sell',​
         orderType: isMarket ? 'Market' : 'Limit',​
         qty: String(intent.qty),​
         orderLinkId: intent.clientOrderId,​
         timeInForce: intent.orderType === 'LIMIT_POST_ONLY' ? 'PostOnly' : 'IOC',​
         reduceOnly: intent.engine === 'EXIT_SUPERVISOR',​
     };​
​
     if (!isMarket && intent.limitPrice > 0) {​
       body.price = String(intent.limitPrice);​
     }​
​
     try {​

      const res = await this.signedRestRequest<{​
        retCode: number;​
       retMsg: string;​
       result?: { orderId?: string };​
      }>('POST', '/v5/order/create', body);​
​
      if (res.retCode !== 0) {​
        tracked.state = 'REJECTED';​
        this.router.publishFromRole('OMS', 'LiveOmsAdapter', 'fabric:error', {​
          source: 'LiveOmsAdapter',​
          error: new Error(`Order rejected (${res.retCode}): ${res.retMsg}`),​
          tsMs: Date.now(),​
        });​
        return;​
      }​
​
      if (tracked.state === 'PENDING_NEW') tracked.state = 'ACKED';​
      tracked.exchangeOrderId = res.result?.orderId;​
​
      if (intent.orderType === 'LIMIT_POST_ONLY') {​
        const timeout = setTimeout(​
          () => void this.cancelIfOpen(intent.symbol, intent.clientOrderId),​
            this.config.postOnlyFallbackMs ?? 2_500​
          );​
          timeout.unref();​
        }​
      } catch (error) {​
        tracked.state = 'REJECTED';​
        this.router.publishFromRole('OMS', 'LiveOmsAdapter', 'fabric:error', {​
          source: 'LiveOmsAdapter',​
          error,​
          tsMs: Date.now(),​
        });​
      }​
    }​
​
    private async cancelIfOpen(symbol: string, clientOrderId: string): Promise<void> {​
     const tracked = this.orders.get(clientOrderId);​
     if (!tracked || ['FILLED', 'CANCELED', 'REJECTED'].includes(tracked.state)) return;​
​
     try {​
      await this.signedRestRequest('POST', '/v5/order/cancel', {​
        category: 'linear',​

           symbol,​
           orderLinkId: clientOrderId,​
       });​
     } catch {​
       // Resolved on next WS update or REST reconciliation​
      }​
    }​
​
    private async refreshDeadMansSwitch(): Promise<void> {​
     try {​
       await this.signedRestRequest('POST', '/v5/order/disconnected-cancel-all', {​
           product: 'FUTURE',​
           timeWindow: this.config.deadManTimeoutSec ?? 10,​
       });​
      } catch (error) {​
        this.router.publishFromRole('OMS', 'LiveOmsAdapter', 'fabric:error', {​
          source: 'LiveOmsAdapter:DeadManSwitch',​
          error,​
          tsMs: Date.now(),​
        });​
      }​
    }​
​
    public async reconcilePositionsWithExchange(): Promise<void> {​
     try {​
       const res = await this.signedRestRequest<{​
           retCode: number;​
        result?: {​
          list?: Array<{​
            symbol: string;​
            side: 'Buy' | 'Sell' | 'None';​
            size: string;​
            avgPrice: string;​
            unrealisedPnl: string;​
          }>;​
        };​
       }>('GET', '/v5/position/list?category=linear&settleCoin=USDT');​
​
       if (res.retCode !== 0 || !res.result?.list) return;​
​
       const snapshots: ExchangePositionSnapshot[] = res.result.list​
           .filter((item) => item.side !== 'None' && Number(item.size) > 0)​
           .map((item) => ({​

         symbol: item.symbol,​
         side: item.side === 'Buy' ? 'BUY' : 'SELL',​
         qty: Number(item.size),​
         entryPrice: Number(item.avgPrice),​
         unrealizedPnlUsd: Number(item.unrealisedPnl || 0),​
        }));​
​
       this.ledger.reconcileExchangeSnapshot(snapshots);​
     } catch (error) {​
      this.router.publishFromRole('OMS', 'LiveOmsAdapter', 'fabric:error', {​
        source: 'LiveOmsAdapter:Reconcile',​
          error,​
          tsMs: Date.now(),​
        });​
      }​
    }​
​
    private connectPrivateWebSocket(): void {​
     if (this.isShuttingDown) return;​
​
     const ws = new WebSocket(this.config.privateWsUrl);​
     this.ws = ws;​
​
     ws.on('open', () => {​
      this.reconnectAttempts = 0;​
      const expires = Date.now() + 10_000;​
      const signature = createHmac('sha256', this.config.apiSecret)​
        .update(`GET/realtime${expires}`)​
        .digest('hex');​
​
      ws.send(JSON.stringify({ op: 'auth', args: [this.config.apiKey, expires, signature] }));​
​
      this.pingTimer = setInterval(() => {​
        if (ws.readyState === WebSocket.OPEN) ws.send(JSON.stringify({ op: 'ping' }));​
       }, 15_000);​
       this.pingTimer.unref();​
     });​
​
     ws.on('message', (raw) => {​
      try {​
        const msg = JSON.parse(raw.toString()) as {​
         op?: string;​
         success?: boolean;​

      topic?: string;​
      data?: Array<Record<string, string>>;​
    };​
​
    if (msg.op === 'auth' && msg.success) {​
      ws.send(JSON.stringify({ op: 'subscribe', args: ['order', 'execution'] }));​
      void this.reconcilePositionsWithExchange();​
      return;​
    }​
​
    if (msg.topic === 'order' && Array.isArray(msg.data)) {​
     for (const u of msg.data) {​
      const tracked = u.orderLinkId ? this.orders.get(u.orderLinkId) : undefined;​
      if (!tracked) continue;​
      if (u.orderStatus === 'New') tracked.state = 'ACKED';​
      else if (u.orderStatus === 'PartiallyFilled') tracked.state = 'PARTIAL';​
      else if (u.orderStatus === 'Filled') tracked.state = 'FILLED';​
      else if (u.orderStatus === 'Cancelled' || u.orderStatus === 'PartiallyFilledCanceled') {​
          tracked.state = 'CANCELED';​
        } else if (u.orderStatus === 'Rejected') tracked.state = 'REJECTED';​
      }​
    } else if (msg.topic === 'execution' && Array.isArray(msg.data)) {​
      for (const ex of msg.data) {​
        const clientOrderId = ex.orderLinkId;​
        const execQty = Number(ex.execQty || 0);​
        const execPrice = Number(ex.execPrice || 0);​
        if (!clientOrderId || execQty <= 0 || execPrice <= 0) continue;​
​
      const tracked = this.orders.get(clientOrderId);​
      if (tracked) {​
        tracked.cumFilledQty += execQty;​
        tracked.cumFeeUsd += Number(ex.execFee || 0);​
        tracked.state =​
          tracked.cumFilledQty >= tracked.intent.qty - 1e-6 ? 'FILLED' : 'PARTIAL';​
      }​
​
      const frame = this.router.getLatestFrame(ex.symbol);​
      const midPriceAtFill = frame ? (frame.bestBid + frame.bestAsk) / 2 : execPrice;​
​
      this.router.publishFromRole('OMS', 'LiveOmsAdapter', 'oms:fill', {​
        clientOrderId,​
        positionId: tracked?.intent.signalId ?? `POS_${ex.symbol}`,​
        symbol: ex.symbol,​

           side: ex.side === 'Buy' ? 'BUY' : 'SELL',​
           filledQty: execQty,​
           avgFillPrice: execPrice,​
           feeUsd: Number(ex.execFee || 0),​
               midPriceAtFill,​
               tsMs: Number(ex.execTime || Date.now()),​
             });​
           }​
         }​
       } catch (error) {​
         this.router.publishFromRole('OMS', 'LiveOmsAdapter', 'fabric:error', {​
           source: 'LiveOmsAdapter:WSParse',​
           error,​
           tsMs: Date.now(),​
         });​
       }​
     });​
​
      ws.on('close', () => {​
        if (this.pingTimer) clearInterval(this.pingTimer);​
        if (this.isShuttingDown) return;​
        const backoffMs = Math.min(15_000, 500 * 2 ** this.reconnectAttempts++);​
        const timer = setTimeout(() => this.connectPrivateWebSocket(), backoffMs);​
        timer.unref();​
      });​
    }​
​
    private async signedRestRequest<T>(​
      method: 'GET' | 'POST',​
      pathWithQuery: string,​
      body?: Record<string, unknown>​
    ): Promise<T> {​
      const timestamp = String(Date.now());​
      const recvWindow = String(this.config.recvWindowMs ?? 5_000);​
      const [path, query = ''] = pathWithQuery.split('?');​
      const payloadStr = method === 'GET' ? query : JSON.stringify(body ?? {});​
      const signature = createHmac('sha256', this.config.apiSecret)​
       .update(`${timestamp}${this.config.apiKey}${recvWindow}${payloadStr}`)​
       .digest('hex');​
​
     const response = await fetch(`${this.config.restBaseUrl}${pathWithQuery}`, {​
      method,​
      headers: {​

          'Content-Type': 'application/json',​
          'X-BAPI-API-KEY': this.config.apiKey,​
          'X-BAPI-SIGN': signature,​
          'X-BAPI-TIMESTAMP': timestamp,​
          'X-BAPI-RECV-WINDOW': recvWindow,​
         },​
         body: method === 'POST' ? payloadStr : undefined,​
     });​
​
     if (!response.ok) throw new Error(`HTTP ${response.status} on ${method} ${path}`);​
     return (await response.json()) as T;​
    }​
​
    public shutdown(): void {​
     this.isShuttingDown = true;​
     if (this.deadManTimer) clearInterval(this.deadManTimer);​
     if (this.reconcileTimer) clearInterval(this.reconcileTimer);​
     if (this.pingTimer) clearInterval(this.pingTimer);​
     if (this.ws) {​
         this.ws.close();​
         this.ws = null;​
    }​
  }​
}​

​
    /**​
     * Fail-closed startup sequence:​
     * 1. Hydrate SQLite WAL equity & daily PnL into UnifiedRiskGovernor​
     * 2. Re-broadcast persisted open positions to ExitSupervisor & RiskGovernor​
     * 3. Connect OMS & reconcile against live exchange snapshot before enabling strategy ingress​

    public async boot(): Promise<void> {​
     this.riskGovernor.reconcileStartupState(​
         this.ledger.getEquity(),​

     * Express/Fastify-compatible unbuffered SSE endpoint handler for ExecutionTelemetryPanel.tsx.​

    public handleTelemetrySse = (req: Request, res: Response): void => {​
     res.writeHead(200, {​
       'Content-Type': 'text/event-stream',​
       'Cache-Control': 'no-cache, no-transform',​
       Connection: 'keep-alive',​
       'X-Accel-Buffering': 'no',​
     });​
​
     const listeners = STREAMED_TELEMETRY_TOPICS.map((topic) => {​
         const fn = (payload: unknown) => {​
           this.sseSeq += 1;​
         res.write(`id: ${this.sseSeq}\nevent: ${topic}\ndata: ${JSON.stringify(payload)}\n\n`);​
       };​
       this.router.onTyped(topic, fn as never);​
       return { topic, fn };​
     });​
​
     const heartbeat = setInterval(() => res.write(': heartbeat\n\n'), 15_000);​

   ​
         req.on('close', () => {​
           clearInterval(heartbeat);​
           for (const { topic, fn } of listeners) {​
             this.router.off(topic, fn);​
           }​
         });​
       };​
