function clamp(val: number, min: number, max: number): number {​
    return Math.max(min, Math.min(max, val));​
}​

function updateEma(prev: number, price: number, period: number): number {​
    const k = 2 / (period + 1);​
    return prev <= 0 ? price : price * k + prev * (1 - k);​
}​

class Float64RingBuffer {​
    private readonly buf: Float64Array;​
    private readonly capacity: number;​
    private head = 0;​
    private count = 0;​
​
    constructor(capacity: number) {​
     this.capacity = capacity;​
     this.buf = new Float64Array(capacity);​
    }​
​
    public get length(): number {​
     return this.count;​
    }​
​
    public push(value: number): void {​
     this.buf[this.head] = value;​
     this.head = (this.head + 1) % this.capacity;​
     if (this.count < this.capacity) {​
         this.count += 1;​
     }​

    }​
​
    /** Returns element `ago` steps back (0 = most recent). */​
    public get(ago: number): number {​
     if (ago < 0 || ago >= this.count) return Number.NaN;​
     const idx = (this.head - 1 - ago + this.capacity) % this.capacity;​
     return this.buf[idx];​
    }​
​
    public mean(window = this.count): number {​
     const n = Math.min(window, this.count);​
     if (n <= 0) return 0;​
     let sum = 0;​
     for (let i = 0; i < n; i++) {​
        sum += this.get(i);​
      }​
      return sum / n;​
    }​
​
    public stdDev(window = this.count): number {​
     const n = Math.min(window, this.count);​
     if (n < 2) return 0;​
     const avg = this.mean(n);​
     let sq = 0;​
     for (let i = 0; i < n; i++) {​
         const d = this.get(i) - avg;​
        sq += d * d;​
      }​
      return Math.sqrt(sq / (n - 1));​
    }​
​
    public min(window = this.count): number {​
     const n = Math.min(window, this.count);​
     if (n <= 0) return 0;​
     let m = Number.POSITIVE_INFINITY;​
     for (let i = 0; i < n; i++) {​
         const v = this.get(i);​
         if (v < m) m = v;​
     }​
     return m;​
    }​
​
    public max(window = this.count): number {​

     const n = Math.min(window, this.count);​
     if (n <= 0) return 0;​
     let m = Number.NEGATIVE_INFINITY;​
     for (let i = 0; i < n; i++) {​
         const v = this.get(i);​
         if (v > m) m = v;​
     }​
     return m;​
    }​
​
    public median(window = this.count): number {​
     const n = Math.min(window, this.count);​
     if (n <= 0) return 0;​
     const tmp: number[] = new Array(n);​
     for (let i = 0; i < n; i++) tmp[i] = this.get(i);​
      tmp.sort((a, b) => a - b);​
      const mid = Math.floor(n / 2);​
      return n % 2 === 0 ? (tmp[mid - 1] + tmp[mid]) * 0.5 : tmp[mid];​
    }​
​
    public percentileRank(value: number, window = this.count): number {​
     const n = Math.min(window, this.count);​
     if (n <= 0) return 50;​
     let below = 0;​
     for (let i = 0; i < n; i++) {​
         if (this.get(i) <= value) below += 1;​
     }​
     return (below / n) * 100;​
  }​
}​

export class FeatureVectorBuilder {​
    private readonly states = new Map<string, SymbolFeatureState>();​
​
    public registerSymbol(​
      symbolId: number,​
      symbol: string,​
      sleeve: AssetSleeve​
    ): void {​
      if (!this.states.has(symbol)) {​
        this.states.set(symbol, new SymbolFeatureState(symbolId, symbol, sleeve));​
      }​
    }​
​
    private getOrThrow(symbol: string): SymbolFeatureState {​
     const state = this.states.get(symbol);​
     if (!state) {​
         throw new Error(`FEATURE_BUILDER_UNREGISTERED_SYMBOL: ${symbol}`);​
     }​
     return state;​
    }​
​
    public onBarClose(symbol: string, bar: OhlcvBarInput): Float64Array {​
     const s = this.getOrThrow(symbol);​
     const f = s.features;​

    const { open, high, low, close, volume, timestampMs } = bar;​
​
    s.feedTimestampsMs.L1_TICK = timestampMs;​
    if (s.sleeve === 'stocks_etf') {​
      s.feedTimestampsMs.EQUITIES_FEED = timestampMs;​
    }​
​
    if (bar.isHigherTimeframe) {​
      s.htfCloseRing.push(close);​
      if (s.htfCloseRing.length >= 10) {​
        const htfFast = s.htfCloseRing.mean(5);​
        const htfSlow = s.htfCloseRing.mean(15);​
        f[FeatureSlot.HTF_BIAS] =​
          close > htfFast && htfFast > htfSlow​
           ? 1​
           : close < htfFast && htfFast < htfSlow​
             ? -1​
             : 0;​
      }​
      return f;​
    }​
​
    if (bar.isSessionOpen || s.vwapVolSum === 0) {​
      s.resetSessionAccumulators(close);​
    }​
​
    const prevClose = s.closeRing.length > 0 ? s.closeRing.get(0) : close;​
    const prevHigh = s.highRing.length > 0 ? s.highRing.get(0) : high;​
    const prevLow = s.lowRing.length > 0 ? s.lowRing.get(0) : low;​
​
    s.openRing.push(open);​
    s.highRing.push(high);​
    s.lowRing.push(low);​
    s.closeRing.push(close);​
    s.volumeRing.push(volume);​
​
    s.ema9 = updateEma(s.ema9, close, 9);​
    s.ema12 = updateEma(s.ema12, close, 12);​
    s.ema20 = updateEma(s.ema20, close, 20);​
    s.ema21 = updateEma(s.ema21, close, 21);​
    s.ema26 = updateEma(s.ema26, close, 26);​
    s.ema55 = updateEma(s.ema55, close, 55);​
​

    f[FeatureSlot.EMA_9] = s.ema9;​
    f[FeatureSlot.EMA_21] = s.ema21;​
    f[FeatureSlot.EMA_55] = s.ema55;​
​
    const macdLine = s.ema12 - s.ema26;​
    s.macdSignal = updateEma(s.macdSignal, macdLine, 9);​
    const macdHist = macdLine - s.macdSignal;​
    f[FeatureSlot.MACD_HIST] = macdHist;​
    f[FeatureSlot.MACD_HIST_SLOPE] = macdHist - s.prevMacdHist;​
    s.prevMacdHist = macdHist;​
​
    if (s.highRing.length >= 20) {​
      f[FeatureSlot.DONCHIAN_20_HIGH] = s.highRing.max(20);​
      f[FeatureSlot.DONCHIAN_20_LOW] = s.lowRing.min(20);​
    }​
​
    const tr = Math.max(​
      high - low,​
      Math.abs(high - prevClose),​
      Math.abs(low - prevClose)​
    );​
    s.atr14 = s.atr14 <= 0 ? tr : (s.atr14 * 13 + tr) / 14;​
    s.atrRing.push(s.atr14);​
​
    const upMove = high - prevHigh;​
    const downMove = prevLow - low;​
    const plusDm = upMove > downMove && upMove > 0 ? upMove : 0;​
    const minusDm = downMove > upMove && downMove > 0 ? downMove : 0;​
​
    s.smoothTr = s.smoothTr <= 0 ? tr : s.smoothTr - s.smoothTr / 14 + tr;​
    s.smoothPlusDm =​
     s.smoothPlusDm <= 0​
      ? plusDm​
      : s.smoothPlusDm - s.smoothPlusDm / 14 + plusDm;​
    s.smoothMinusDm =​
     s.smoothMinusDm <= 0​
      ? minusDm​
      : s.smoothMinusDm - s.smoothMinusDm / 14 + minusDm;​
​
    const plusDi = s.smoothTr > 0 ? (100 * s.smoothPlusDm) / s.smoothTr : 0;​
    const minusDi = s.smoothTr > 0 ? (100 * s.smoothMinusDm) / s.smoothTr : 0;​
    const diSum = plusDi + minusDi;​
    const dx = diSum > 0 ? (100 * Math.abs(plusDi - minusDi)) / diSum : 0;​

    s.adx14 = s.adx14 <= 0 ? dx : (s.adx14 * 13 + dx) / 14;​
​
    f[FeatureSlot.ADX_14] = s.adx14;​
    f[FeatureSlot.PLUS_DI_14] = plusDi;​
    f[FeatureSlot.MINUS_DI_14] = minusDi;​
​
    const hl2 = (high + low) * 0.5;​
    const basicUpper = hl2 + 3.0 * s.atr14;​
    const basicLower = hl2 - 3.0 * s.atr14;​
    s.supertrendUpper =​
      basicUpper < s.supertrendUpper || prevClose > s.supertrendUpper​
        ? basicUpper​
        : s.supertrendUpper || basicUpper;​
    s.supertrendLower =​
      basicLower > s.supertrendLower || prevClose < s.supertrendLower​
        ? basicLower​
        : s.supertrendLower || basicLower;​
    if (close > s.supertrendUpper) s.supertrendDir = 1;​
    else if (close < s.supertrendLower) s.supertrendDir = -1;​
    f[FeatureSlot.SUPERTREND_DIR] = s.supertrendDir;​
​
    const typicalPrice = (high + low + close) / 3;​
    const effVol = Math.max(volume, 1e-6);​
    s.vwapPvSum += typicalPrice * effVol;​
    s.vwapVolSum += effVol;​
    s.vwapP2vSum += typicalPrice * typicalPrice * effVol;​
​
    const vwap = s.vwapPvSum / s.vwapVolSum;​
    const variance = Math.max(0, s.vwapP2vSum / s.vwapVolSum - vwap * vwap);​
    const vwapStd = Math.sqrt(variance);​
    f[FeatureSlot.SESSION_VWAP] = vwap;​
    f[FeatureSlot.VWAP_SLOPE] = s.prevVwap > 0 ? vwap - s.prevVwap : 0;​
    f[FeatureSlot.VWAP_SIGMA_DIST] =​
     vwapStd > 1e-8 ? (close - vwap) / vwapStd : 0;​
    s.prevVwap = vwap;​
​
    const pocBin = clamp(​
     Math.floor((typicalPrice - s.pocAnchorLow) / s.pocBinStep),​
     0,​
     63​
    );​
    s.pocBins[pocBin] += effVol;​
    let maxBinIdx = 0;​

    for (let i = 1; i < 64; i++) {​
     if (s.pocBins[i] > s.pocBins[maxBinIdx]) maxBinIdx = i;​
    }​
    const pocPrice = s.pocAnchorLow + (maxBinIdx + 0.5) * s.pocBinStep;​
    f[FeatureSlot.POC_DIST_ATR] =​
     s.atr14 > 0 ? (close - pocPrice) / s.atr14 : 0;​
​
    if (s.closeRing.length >= 20) {​
     const n = 20;​
     const xMean = (n - 1) * 0.5;​
     const yMean = s.closeRing.mean(n);​
     let sxx = 0;​
     let sxy = 0;​
     let syy = 0;​
     for (let i = 0; i < n; i++) {​
         const x = n - 1 - i;​
         const y = s.closeRing.get(i);​
        sxx += (x - xMean) * (x - xMean);​
        sxy += (x - xMean) * (y - yMean);​
        syy += (y - yMean) * (y - yMean);​
      }​
      const slope = sxx > 0 ? sxy / sxx : 0;​
      const r2 = sxx > 0 && syy > 0 ? (sxy * sxy) / (sxx * syy) : 0;​
      f[FeatureSlot.OLS_SLOPE_BPS] = yMean > 0 ? (slope / yMean) * 10_000 : 0;​
      f[FeatureSlot.OLS_R_SQUARED] = clamp(r2, 0, 1);​
    }​
​
    if (s.highRing.length >= 52) {​
     const tenkan = (s.highRing.max(9) + s.lowRing.min(9)) * 0.5;​
     const kijun = (s.highRing.max(26) + s.lowRing.min(26)) * 0.5;​
     const spanA = (tenkan + kijun) * 0.5;​
     const spanB = (s.highRing.max(52) + s.lowRing.min(52)) * 0.5;​
      f[FeatureSlot.ICHIMOKU_TENKAN] = tenkan;​
      f[FeatureSlot.ICHIMOKU_KIJUN] = kijun;​
      f[FeatureSlot.ICHIMOKU_KUMO_TOP] = Math.max(spanA, spanB);​
    }​
​
    if (bar.etMinutesSinceMidnight !== undefined) {​
     const mins = bar.etMinutesSinceMidnight;​
     if (mins >= 570 && mins < 585) {​
       s.orbHigh = s.orbHigh <= 0 ? high : Math.max(s.orbHigh, high);​
       s.orbLow = s.orbLow <= 0 ? low : Math.min(s.orbLow, low);​
     }​

      f[FeatureSlot.EQUITIES_ORB_HIGH] = s.orbHigh;​
      f[FeatureSlot.EQUITIES_ORB_LOW] = s.orbLow;​
      f[FeatureSlot.POST_OPEN_VOL_STABLE] = mins >= 590 && mins <= 955 ? 1 : 0;​
    }​
​
    if (s.highRing.length >= 12) {​
     const recentHigh = s.highRing.max(6);​
     const priorHigh = Math.max(​
        s.highRing.get(6),​
        s.highRing.get(7),​
        s.highRing.get(8),​
        s.highRing.get(9),​
        s.highRing.get(10),​
        s.highRing.get(11)​
      );​
      const recentLow = s.lowRing.min(6);​
      const priorLow = Math.min(​
        s.lowRing.get(6),​
        s.lowRing.get(7),​
        s.lowRing.get(8),​
        s.lowRing.get(9),​
        s.lowRing.get(10),​
        s.lowRing.get(11)​
      );​
      f[FeatureSlot.SWING_STRUCTURE_CODE] =​
        recentHigh > priorHigh && recentLow > priorLow​
          ? 1​
          : recentHigh < priorHigh && recentLow < priorLow​
            ? -1​
            : 0;​
    }​
​
    const rsi2 = this.computeRsi(s.closeRing, 2);​
    const rsi14 = this.computeRsi(s.closeRing, 14);​
    s.rsi14Ring.push(rsi14);​
    f[FeatureSlot.CONNORS_RSI_2] = rsi2;​
​
    if (s.closeRing.length >= 10 && s.rsi14Ring.length >= 10) {​
     const priceNow = close;​
     const pricePast = s.closeRing.get(8);​
     const rsiNow = rsi14;​
     const rsiPast = s.rsi14Ring.get(8);​
     f[FeatureSlot.RSI14_BULL_DIV] =​

       priceNow < pricePast && rsiNow > rsiPast + 4 && rsiNow < 42 ? 1 : 0;​
      f[FeatureSlot.RSI14_BEAR_DIV] =​
       priceNow > pricePast && rsiNow < rsiPast - 4 && rsiNow > 58 ? 1 : 0;​
    }​
​
    if (s.closeRing.length >= 20) {​
     const sma20 = s.closeRing.mean(20);​
     const sd20 = s.closeRing.stdDev(20);​
     const bbUpper = sma20 + 2 * sd20;​
     const bbLower = sma20 - 2 * sd20;​
     const bbWidth = bbUpper - bbLower;​
     f[FeatureSlot.BB_PERCENT_B] =​
      bbWidth > 1e-8 ? (close - bbLower) / bbWidth : 0.5;​
​
     const kcHalfWidth = 2.25 * s.atr14;​
     f[FeatureSlot.KELTNER_POS] =​
      kcHalfWidth > 1e-8 ? (close - s.ema20) / kcHalfWidth : 0;​
​
     const kcStdUpper = s.ema20 + 1.5 * s.atr14;​
     const kcStdLower = s.ema20 - 1.5 * s.atr14;​
     const inSqueezeNow = bbUpper < kcStdUpper && bbLower > kcStdLower;​
     const momentum = close - (s.highRing.max(20) + s.lowRing.min(20)) * 0.5;​
      f[FeatureSlot.TTM_SQUEEZE_RELEASE] =​
       !inSqueezeNow && s.atrRing.length >= 5 && s.atr14 > s.atrRing.get(1)​
         ? momentum > 0​
           ? 1​
           : -1​
         : 0;​
    }​
​
    if (s.closeRing.length >= 21) {​
     const retNow = prevClose > 0 ? (close - prevClose) / prevClose : 0;​
     const retStd = Math.max(s.atr14 / Math.max(close, 1e-8), 1e-5);​
      f[FeatureSlot.RETURN_Z_SCORE] = retNow / retStd;​
    }​
​
    if (s.highRing.length >= 4) {​
     const c2High = s.highRing.get(3);​
     const c2Low = s.lowRing.get(3);​
     const c0Low = s.lowRing.get(1);​
     const c0High = s.highRing.get(1);​
     if (c0Low > c2High) {​
      const fvgMid = (c0Low + c2High) * 0.5;​

        f[FeatureSlot.FVG_EQUILIBRIUM_SIGNAL] =​
         low <= fvgMid && close > fvgMid ? 1 : 0;​
      } else if (c0High < c2Low) {​
        const fvgMid = (c0High + c2Low) * 0.5;​
        f[FeatureSlot.FVG_EQUILIBRIUM_SIGNAL] =​
         high >= fvgMid && close < fvgMid ? -1 : 0;​
      } else {​
        f[FeatureSlot.FVG_EQUILIBRIUM_SIGNAL] = 0;​
      }​
    }​
​
    if (s.lowRing.length >= 15) {​
     let prior14Low = Number.POSITIVE_INFINITY;​
     let prior14High = Number.NEGATIVE_INFINITY;​
     for (let i = 1; i <= 14; i++) {​
        prior14Low = Math.min(prior14Low, s.lowRing.get(i));​
        prior14High = Math.max(prior14High, s.highRing.get(i));​
      }​
      if (low < prior14Low && close > prior14Low) {​
        f[FeatureSlot.SWEEP_RECLAIM_SIGNAL] = 1;​
      } else if (high > prior14High && close < prior14High) {​
        f[FeatureSlot.SWEEP_RECLAIM_SIGNAL] = -1;​
      } else {​
        f[FeatureSlot.SWEEP_RECLAIM_SIGNAL] = 0;​
      }​
    }​
​
    const medianAtr = s.atrRing.median(60);​
    f[FeatureSlot.ATR_EXPANSION_RATIO] =​
     medianAtr > 0 ? s.atr14 / medianAtr : 1.0;​
    f[FeatureSlot.ATR_PERCENTILE_RANK] = s.atrRing.percentileRank(s.atr14, 120);​
​
    const barRange = Math.max(high - low, 1e-8);​
    f[FeatureSlot.PARKINSON_EFFICIENCY] = clamp(​
      (close - open) / barRange,​
      -1,​
      1​
    );​
​
    const logRet = prevClose > 0 ? Math.log(close / prevClose) : 0;​
    const sqRet = logRet * logRet;​
    s.prevGarchVariance = s.garchVariance || sqRet;​
    s.garchVariance =​

         1e-7 + 0.12 * sqRet + 0.85 * (s.garchVariance || sqRet);​
     f[FeatureSlot.GARCH_COND_VOL_SLOPE] =​
      s.prevGarchVariance > 0​
       ? (s.garchVariance - s.prevGarchVariance) / s.prevGarchVariance​
       : 0;​
​
     f[FeatureSlot.TAIL_KURTOSIS_DROP] =​
      Math.abs(f[FeatureSlot.RETURN_Z_SCORE]) < 1.2 &&​
      f[FeatureSlot.ATR_EXPANSION_RATIO] > 1.2​
       ? 1​
       : 0;​
​
     f[FeatureSlot.EXTENSION_FROM_EMA_ATR] =​
      s.atr14 > 0 ? (close - s.ema21) / s.atr14 : 0;​
​
     s.barStartBuyVol = 0;​
     s.barStartSellVol = 0;​
​
     return f;​
    }​

    public onL2BookUpdate(symbol: string, snap: L2BookSnapshotInput): Float64Array {​
     const s = this.getOrThrow(symbol);​
     const f = s.features;​
     s.feedTimestampsMs.L2_BOOK = snap.timestampMs;​
​
     const bestBid = snap.bids[0];​
     const bestAsk = snap.asks[0];​
     if (!bestBid || !bestAsk || bestBid.price <= 0 || bestAsk.price <= 0) {​
         return f;​
     }​
​
     const mid = (bestBid.price + bestAsk.price) * 0.5;​
     const spreadBps = ((bestAsk.price - bestBid.price) / mid) * 10_000;​
     s.spreadRing.push(spreadBps);​
​
     const medianSpread30m = Math.max(s.spreadRing.median(180), 0.1);​
     const spreadMult = spreadBps / medianSpread30m;​
     f[FeatureSlot.SPREAD_VS_30M_MEDIAN_MULT] = spreadMult;​
     f[FeatureSlot.SPREAD_COMPRESSION_RATIO] = clamp(1 - spreadMult, -2, 1);​
​
     const denom = bestBid.size + bestAsk.size;​
     const microPrice =​

     denom > 0​
      ? (bestBid.price * bestAsk.size + bestAsk.price * bestBid.size) / denom​
      : mid;​
    f[FeatureSlot.MICROPRICE_SKEW_BPS] = ((microPrice - mid) / mid) * 10_000;​
​
    let bidDepletion = 0;​
    let askDepletion = 0;​
    if (s.prevBestBidPrice === bestBid.price && s.prevBestBidSize > 0) {​
     const delta = s.prevBestBidSize - bestBid.size;​
     if (delta > 0) bidDepletion = delta / s.prevBestBidSize;​
     else if (delta < 0) s.bidReloadCount += 1;​
    }​
    if (s.prevBestAskPrice === bestAsk.price && s.prevBestAskSize > 0) {​
     const delta = s.prevBestAskSize - bestAsk.size;​
     if (delta > 0) askDepletion = delta / s.prevBestAskSize;​
     else if (delta < 0) s.askReloadCount += 1;​
    }​
    f[FeatureSlot.QUEUE_DEPLETION_ASYM] = clamp(​
      askDepletion - bidDepletion,​
      -1,​
      1​
    );​
​
    const totalReloads = s.bidReloadCount + s.askReloadCount;​
    const reloadSkew =​
     totalReloads > 0​
      ? (s.bidReloadCount - s.askReloadCount) / totalReloads​
      : 0;​
    f[FeatureSlot.BOOK_RELOAD_SKEW] = reloadSkew;​
    f[FeatureSlot.ICEBERG_RELOAD_SIDE] =​
     totalReloads >= 6 && reloadSkew >= 0.55​
      ? 1​
      : totalReloads >= 6 && reloadSkew <= -0.55​
        ? -1​
        : 0;​
​
    const topN = Math.min(5, snap.bids.length, snap.asks.length);​
    let bidSum = 0;​
    let askSum = 0;​
    const sizes: number[] = [];​
    for (let i = 0; i < topN; i++) {​
     bidSum += snap.bids[i].size;​
     askSum += snap.asks[i].size;​

      sizes.push(snap.bids[i].size, snap.asks[i].size);​
    }​
    sizes.sort((a, b) => a - b);​
    const medianDepth =​
      sizes.length > 0 ? Math.max(sizes[Math.floor(sizes.length / 2)], 1e-6) : 1;​
​
    let maxBidWall = 0;​
    let maxAskWall = 0;​
    for (let i = 0; i < topN; i++) {​
      maxBidWall = Math.max(maxBidWall, snap.bids[i].size);​
      maxAskWall = Math.max(maxAskWall, snap.asks[i].size);​
    }​
    f[FeatureSlot.BID_WALL_DEPTH_RATIO] = maxBidWall / medianDepth;​
​
    const ofiL5 = bidSum + askSum > 0 ? (bidSum - askSum) / (bidSum + askSum) : 0;​
    f[FeatureSlot.OFI_L5_ACCEL] = ofiL5 - s.prevOfiL5;​
    s.prevOfiL5 = ofiL5;​
​
    if (s.cvdDelta1m > 20_000 && maxAskWall / medianDepth >= 3.0 && ofiL5 > 0.1) {​
      f[FeatureSlot.WALL_ABSORPTION_SIGNAL] = 1;​
    } else if (​
      s.cvdDelta1m < -20_000 &&​
      maxBidWall / medianDepth >= 3.0 &&​
      ofiL5 < -0.1​
    ) {​
      f[FeatureSlot.WALL_ABSORPTION_SIGNAL] = -1;​
    } else {​
      f[FeatureSlot.WALL_ABSORPTION_SIGNAL] = 0;​
    }​
​
    if (topN >= 3) {​
     const bidSlope =​
       (snap.bids[topN - 1].size - snap.bids[0].size) / Math.max(bidSum, 1e-6);​
      const askSlope =​
       (snap.asks[topN - 1].size - snap.asks[0].size) / Math.max(askSum, 1e-6);​
      f[FeatureSlot.BOOK_SLOPE_CONVEXITY_DIFF] = bidSlope - askSlope;​
    }​
​
    if (​
      snap.recentCancels !== undefined &&​
      snap.recentFills !== undefined &&​
      snap.recentFills > 0​
    ) {​

       f[FeatureSlot.CANCEL_TO_FILL_RATIO] =​
        snap.recentCancels / snap.recentFills;​
     }​
​
     if (snap.leaderVenueMidPrice && snap.leaderVenueMidPrice > 0) {​
       f[FeatureSlot.LEAD_VENUE_MID_DIFF_BPS] =​
        ((snap.leaderVenueMidPrice - mid) / mid) * 10_000;​
     }​
​
     if (snap.oracleMarkPrice && snap.oracleMarkPrice > 0) {​
       f[FeatureSlot.ORACLE_DISAGREEMENT_BPS] =​
        ((mid - snap.oracleMarkPrice) / snap.oracleMarkPrice) * 10_000;​
     }​
​
     if (​
       snap.roundTripFrictionBps !== undefined &&​
       snap.baselineExpectedEdgeBps !== undefined &&​
       snap.baselineExpectedEdgeBps > 0​
     ) {​
       f[FeatureSlot.FRICTION_TO_EDGE_RATIO] =​
        snap.roundTripFrictionBps / snap.baselineExpectedEdgeBps;​
     }​
​
     if (snap.overheadLiquidityWallR !== undefined) {​
       s.overheadLiquidityWallR = snap.overheadLiquidityWallR;​
     }​
​
     s.prevBestBidPrice = bestBid.price;​
     s.prevBestBidSize = bestBid.size;​
     s.prevBestAskPrice = bestAsk.price;​
     s.prevBestAskSize = bestAsk.size;​
​
     return f;​
    }​

    public onTradeTick(symbol: string, trade: TradeTickInput): Float64Array {​
     const s = this.getOrThrow(symbol);​
     const f = s.features;​
     const { timestampMs, price, notionalUsd } = trade;​
     s.feedTimestampsMs.TRADE_TAPE = timestampMs;​
​
     let isBuy: boolean;​
     if (trade.aggressorSide) {​

      isBuy = trade.aggressorSide === 'BUY';​
    } else if (price > s.lastTradePrice) {​
      isBuy = true;​
      s.lastTickDirection = 1;​
    } else if (price < s.lastTradePrice) {​
      isBuy = false;​
      s.lastTickDirection = -1;​
    } else {​
      isBuy = s.lastTickDirection === 1;​
    }​
    s.lastTradePrice = price;​
​
    if (trade.organicVolumeRatio !== undefined) {​
      s.organicVolumeRatio = trade.organicVolumeRatio;​
    }​
​
    s.recentTrades.push({​
     ts: timestampMs,​
     price,​
     notionalUsd,​
     isBuy,​
     isSolanaDex: Boolean(trade.isSolanaDex),​
    });​
​
    const cutoff60s = timestampMs - 60_000;​
    while (s.recentTrades.length > 0 && s.recentTrades[0].ts < cutoff60s) {​
      s.recentTrades.shift();​
    }​
​
    const VPIN_BUCKET_USD = 50_000;​
    s.vpinBucketVol += notionalUsd;​
    if (isBuy) s.vpinBucketBuyVol += notionalUsd;​
    if (s.vpinBucketVol >= VPIN_BUCKET_USD) {​
      const sellVol = s.vpinBucketVol - s.vpinBucketBuyVol;​
      const imbalance =​
        Math.abs(s.vpinBucketBuyVol - sellVol) / s.vpinBucketVol;​
      s.vpinBucketImbalances.push(imbalance);​
      s.vpin = s.vpinBucketImbalances.mean(20);​
      s.vpinBucketVol = 0;​
      s.vpinBucketBuyVol = 0;​
    }​
​
    let buyVol = 0;​

    let sellVol = 0;​
    let dexBuyVol = 0;​
    let dexSellVol = 0;​
    let sweepLevels = 0;​
    let sweepDir = 0;​
    let blockDelta = 0;​
    let maxNotional = 0;​
​
    for (let i = 0; i < s.recentTrades.length; i++) {​
     const t = s.recentTrades[i];​
     if (t.isBuy) {​
       buyVol += t.notionalUsd;​
       if (t.isSolanaDex) dexBuyVol += t.notionalUsd;​
     } else {​
       sellVol += t.notionalUsd;​
       if (t.isSolanaDex) dexSellVol += t.notionalUsd;​
     }​
     if (t.notionalUsd > maxNotional) maxNotional = t.notionalUsd;​
​
     if (timestampMs - t.ts <= 500 && i > 0) {​
      const prev = s.recentTrades[i - 1];​
      if (t.isBuy && t.price > prev.price) {​
          sweepLevels = sweepDir === 1 ? sweepLevels + 1 : 1;​
          sweepDir = 1;​
        } else if (!t.isBuy && t.price < prev.price) {​
          sweepLevels = sweepDir === -1 ? sweepLevels + 1 : 1;​
          sweepDir = -1;​
        }​
      }​
    }​
​
    const blockThreshold = maxNotional * 0.65;​
    for (let i = 0; i < s.recentTrades.length; i++) {​
     const t = s.recentTrades[i];​
     if (t.notionalUsd >= blockThreshold) {​
        blockDelta += t.isBuy ? t.notionalUsd : -t.notionalUsd;​
      }​
    }​
​
    const totalVol = buyVol + sellVol;​
    s.cvdDelta1m = buyVol - sellVol;​
​
    f[FeatureSlot.AGGRESSOR_SWEEP_TICKS] = sweepDir * sweepLevels;​

     f[FeatureSlot.BLOCK_PRINT_DELTA_Z] =​
      totalVol > 0 ? (blockDelta / totalVol) * 4.0 : 0;​
     f[FeatureSlot.TICK_RULE_BUY_RATIO] =​
      totalVol > 0 ? buyVol / totalVol : 0.5;​
     f[FeatureSlot.TAPE_TPS_RATIO] = s.recentTrades.length / 20;​
​
     const ratio =​
      sellVol > 0 ? buyVol / sellVol : buyVol > 0 ? 4.0 : 1.0;​
     const invRatio =​
      buyVol > 0 ? sellVol / buyVol : sellVol > 0 ? 4.0 : 1.0;​
     f[FeatureSlot.FOOTPRINT_IMBALANCE_STACK] =​
      ratio >= 3.0 ? 3 : invRatio >= 3.0 ? -3 : 0;​
​
     const zRet = f[FeatureSlot.RETURN_Z_SCORE];​
     f[FeatureSlot.CLIMAX_EXHAUSTION_FLAG] =​
      f[FeatureSlot.TAPE_TPS_RATIO] >= 3.0 && zRet <= -2.2​
       ? 1​
       : f[FeatureSlot.TAPE_TPS_RATIO] >= 3.0 && zRet >= 2.2​
         ? -1​
         : 0;​
​
     if (s.barStartBuyVol === 0 && s.barStartSellVol === 0) {​
       s.barStartBuyVol = buyVol;​
       s.barStartSellVol = sellVol;​
     }​
     const startDelta = s.barStartBuyVol - s.barStartSellVol;​
     f[FeatureSlot.INTRABAR_DELTA_FLIP] =​
       startDelta < -5_000 && s.cvdDelta1m > 5_000​
        ? 1​
        : startDelta > 5_000 && s.cvdDelta1m < -5_000​
          ? -1​
          : 0;​
​
     const dexTotal = dexBuyVol + dexSellVol;​
     f[FeatureSlot.SOL_DEX_NET_FLOW_RATIO] =​
      dexTotal > 0 ? (dexBuyVol - dexSellVol) / dexTotal : 0;​
​
     return f;​
    }​

​
     f[FeatureSlot.SPL_MINT_REVOKED] = sol.mintAuthorityRevoked ? 1 : 0;​

​
    private computeRsi(ring: Float64RingBuffer, period: number): number {​

      if (ring.length <= period) return 50;​
      let gains = 0;​
      let losses = 0;​
      for (let i = 0; i < period; i++) {​
       const diff = ring.get(i) - ring.get(i + 1);​
       if (diff > 0) gains += diff;​
       else losses -= diff;​
      }​
      if (losses === 0) return gains > 0 ? 100 : 50;​
      const rs = gains / losses;​
      return 100 - 100 / (1 + rs);​
  }​
}​

function calibrateProbability(rawConfidence: number, temperature: number): number {​
    const p = clamp(rawConfidence, 0.01, 0.99);​
    const logit = Math.log(p / (1 - p));​

 const scaledLogit = logit / Math.max(0.25, temperature);​
 return clamp(1 / (1 + Math.exp(-scaledLogit)), 0.05, 0.95);​
}​
​
function sampleGamma(shape: number): number {​
 if (shape < 1) {​
     return sampleGamma(shape + 1) * Math.pow(Math.random(), 1 / shape);​
 }​
 const d = shape - 1 / 3;​
 const c = 1 / Math.sqrt(9 * d);​
 for (;;) {​
     let x = 0;​
     let v = 0;​
     do {​
      const u1 = Math.random();​
      const u2 = Math.random();​
      x =​
       Math.sqrt(-2 * Math.log(Math.max(u1, 1e-12))) *​
        Math.cos(2 * Math.PI * u2);​
      v = 1 + c * x;​
    } while (v <= 0);​
    v = v * v * v;​
    const u = Math.random();​
    if (u < 1 - 0.0331 * x * x * x * x) return d * v;​
    if (Math.log(u) < 0.5 * x * x + d * (1 - v + Math.log(v))) return d * v;​
  }​
}​
​
function sampleBeta(alpha: number, beta: number): number {​
  const x = sampleGamma(Math.max(0.1, alpha));​
  const y = sampleGamma(Math.max(0.1, beta));​
  return x / (x + y);​
}​

// MAIN THREAD SWARM POOL COORDINATOR (isMainThread)​

export class SwarmThreadPool {​
    private readonly workers: Worker[] = [];​
    private readonly shardCount: number;​
    private readonly maxSymbols: number;​
    private readonly sharedBuffer: SharedArrayBuffer;​
    private readonly deadlineMs: number;​
    private readonly pending = new Map<number, PendingBatch>();​
    private nextRequestId = 1;​
    private totalEvaluations = 0;​
    private deadlineTimeouts = 0;​
    private isShuttingDown = false;​
​
    constructor(params: {​
     sharedBuffer: SharedArrayBuffer;​
     maxSymbols: number;​
     shardCount?: number;​
     deadlineMs?: number;​
    }) {​
     this.sharedBuffer = params.sharedBuffer;​

     this.maxSymbols = params.maxSymbols;​
     this.deadlineMs = params.deadlineMs ?? DEFAULT_DEADLINE_MS;​
     const cpuCount = os.cpus().length;​
     this.shardCount =​
       params.shardCount ?? Math.min(4, Math.max(2, cpuCount - 1));​
​
     for (let i = 0; i < this.shardCount; i++) {​
       this.spawnShard(i);​
      }​
    }​
​
    private spawnShard(shardIndex: number): void {​
     const worker = new Worker(new URL(import.meta.url), {​
       workerData: {​
        shardIndex,​
        shardCount: this.shardCount,​
        maxSymbols: this.maxSymbols,​
        sharedBuffer: this.sharedBuffer,​
      } satisfies ShardWorkerInitData,​
      execArgv: process.execArgv,​
     });​
​
     worker.on('message', (msg: WorkerToMainMsg) => {​
      if (msg.type !== 'EVAL_RESULT') return;​
      const batch = this.pending.get(msg.requestId);​
      if (!batch) return;​
​
       if (msg.votes.length > 0) {​
         batch.collectedVotes.push(...msg.votes);​
       }​
       batch.completedShards += 1;​
​
       if (batch.completedShards >= batch.expectedShards) {​
           clearTimeout(batch.timer);​
           this.pending.delete(msg.requestId);​
        batch.resolve({​
          votes: batch.collectedVotes,​
          timedOutShards: 0,​
        });​
      }​
     });​
​
     worker.on('error', (err) => {​

         console.error(`[SwarmThreadPool] Shard ${shardIndex} error:`, err);​
     });​
​
     worker.on('exit', (code) => {​
      if (!this.isShuttingDown && code !== 0) {​
        console.warn(​
            `[SwarmThreadPool] Shard ${shardIndex} exited (${code}); respawning...`​
          );​
          this.spawnShard(shardIndex);​
      }​
     });​
​
     this.workers[shardIndex] = worker;​
    }​
​
    public syncReputationSnapshot(stats: AgentRegimeStats[]): void {​
     const msg: MainToWorkerMsg = { type: 'SYNC_STATS', stats };​
     for (const worker of this.workers) {​
        worker?.postMessage(msg);​
      }​
    }​
​
    public evaluateSymbol(​
      payload: SwarmEvalPayload​
    ): Promise<SwarmConsensusDecision> {​
      const startNs = process.hrtime.bigint();​
      const requestId = this.nextRequestId++;​
      const nowMs = Date.now();​
      this.totalEvaluations += 1;​
​
     return new Promise<SwarmConsensusDecision>((resolve) => {​
         const finalize = (res: {​
          votes: CalibratedAgentVote[];​
          timedOutShards: number;​
         }) => {​
           resolve(this.aggregateVotes(payload, res.votes, startNs));​
         };​
​
         const timer = setTimeout(() => {​
          const batch = this.pending.get(requestId);​
          if (!batch) return;​
          this.pending.delete(requestId);​
          this.deadlineTimeouts += 1;​

        finalize({​
          votes: batch.collectedVotes,​
          timedOutShards: batch.expectedShards - batch.completedShards,​
        });​
      }, this.deadlineMs);​
​
      this.pending.set(requestId, {​
       resolve: finalize,​
       timer,​
       collectedVotes: [],​
       completedShards: 0,​
       expectedShards: this.shardCount,​
      });​
​
      const msg: MainToWorkerMsg = {​
       type: 'EVALUATE',​
        requestId,​
        nowMs,​
        payload,​
      };​
​
      for (const worker of this.workers) {​
         worker.postMessage(msg);​
       }​
      });​
    }​
​
    private aggregateVotes(​
      payload: SwarmEvalPayload,​
      allVotes: CalibratedAgentVote[],​
      startNs: bigint​
    ): SwarmConsensusDecision {​
      const criticVetoes = allVotes.filter(​
        (v) =>​
          v.bucket === 'ADVERSARIAL_CRITIC' &&​
          v.action === 'VETO' &&​
          !v.quarantined​
      );​
      const highSeverityVeto = criticVetoes.some(​
        (v) => v.calibratedWinProb >= 0.88​
      );​
​
     if (criticVetoes.length >= RED_TEAM_VETO_QUORUM || highSeverityVeto) {​

     return {​
      symbol: payload.symbol,​
       sleeve: payload.sleeve,​
       regime: payload.regime,​
       direction: 'HOLD',​
       posteriorWinProb: 0.5,​
       netLogOdds: 0,​
       activeBucketCount: 0,​
       netExpectedEdgeBps: 0,​
       vetoed: true,​
       vetoReasons: criticVetoes.map((v) => `${v.agentId}:${v.reasonCode}`),​
       bucketBreakdown: [],​
       contributingVotes: criticVetoes,​
       latencyUs: Number(process.hrtime.bigint() - startNs) / 1_000,​
      };​
    }​
​
    const byBucket = new Map<EvidenceBucket, CalibratedAgentVote[]>();​
    for (const v of allVotes) {​
     if (v.bucket === 'ADVERSARIAL_CRITIC') continue;​
     const list = byBucket.get(v.bucket) ?? [];​
      list.push(v);​
      byBucket.set(v.bucket, list);​
    }​
​
    let totalCappedLogOdds = 0;​
    let weightedEdgeNum = 0;​
    let weightedEdgeDen = 0;​
    const bucketBreakdown: SwarmConsensusDecision['bucketBreakdown'] = [];​
​
    for (const [bucket, votes] of byBucket.entries()) {​
     const active = votes.filter(​
       (v) => (v.action === 'BUY' || v.action === 'SELL') && v.effectiveWeight > 0​
     );​
     if (active.length === 0) continue;​
​
     const meanLogOdds =​
       active.reduce((sum, v) => sum + v.logOddsContribution, 0) /​
       Math.sqrt(active.length);​
​
     const cappedLogOdds = clamp(​
       meanLogOdds,​
       -MAX_BUCKET_LOG_ODDS,​

       MAX_BUCKET_LOG_ODDS​
     );​
     totalCappedLogOdds += cappedLogOdds;​
​
     const strongest = active.reduce((best, cur) =>​
      Math.abs(cur.logOddsContribution) > Math.abs(best.logOddsContribution)​
           ? cur​
           : best​
     );​
​
     for (const v of active) {​
       weightedEdgeNum += v.expectedEdgeBps * v.effectiveWeight;​
       weightedEdgeDen += v.effectiveWeight;​
     }​
​
      bucketBreakdown.push({​
       bucket,​
       activeVoters: active.length,​
       abstained: votes.length - active.length,​
       rawLogOdds: Number(meanLogOdds.toFixed(4)),​
       cappedLogOdds: Number(cappedLogOdds.toFixed(4)),​
       topReason: strongest.reasonCode,​
      });​
    }​
​
    const direction: 'BUY' | 'SELL' | 'HOLD' =​
     totalCappedLogOdds > 0 ? 'BUY' : totalCappedLogOdds < 0 ? 'SELL' : 'HOLD';​
    const dirSign = direction === 'BUY' ? 1 : direction === 'SELL' ? -1 : 0;​
    const posteriorWinProb = 1 / (1 + Math.exp(-Math.abs(totalCappedLogOdds)));​
​
    const activeBucketCount = bucketBreakdown.filter(​
     (b) => b.cappedLogOdds * dirSign > 0.15​
    ).length;​
​
    const grossEdgeBps =​
     weightedEdgeDen > 0 ? weightedEdgeNum / weightedEdgeDen : 0;​
    const netExpectedEdgeBps = grossEdgeBps - payload.roundTripFrictionBps;​
​
    const approved =​
     direction !== 'HOLD' &&​
     activeBucketCount >= MIN_INDEPENDENT_BUCKETS &&​
     posteriorWinProb >= MIN_POSTERIOR_PROB &&​
     netExpectedEdgeBps > 0;​

​
     return {​
      symbol: payload.symbol,​
      sleeve: payload.sleeve,​
      regime: payload.regime,​
      direction: approved ? direction : 'HOLD',​
      posteriorWinProb: Number(posteriorWinProb.toFixed(4)),​
      netLogOdds: Number(totalCappedLogOdds.toFixed(4)),​
        activeBucketCount,​
        netExpectedEdgeBps: Number(netExpectedEdgeBps.toFixed(2)),​
        vetoed: false,​
        vetoReasons: [],​
        bucketBreakdown,​
        contributingVotes: allVotes.filter(​
          (v) => v.action === direction && v.effectiveWeight > 0​
        ),​
        latencyUs: Number(process.hrtime.bigint() - startNs) / 1_000,​
      };​
    }​
​
    public getMetrics(): {​
     shardCount: number;​
      deadlineMs: number;​
      totalEvaluations: number;​
      deadlineTimeouts: number;​
    } {​
      return {​
        shardCount: this.shardCount,​
        deadlineMs: this.deadlineMs,​
        totalEvaluations: this.totalEvaluations,​
        deadlineTimeouts: this.deadlineTimeouts,​
      };​
    }​
​
    public async shutdown(): Promise<void> {​
     this.isShuttingDown = true;​
     for (const batch of this.pending.values()) {​
      clearTimeout(batch.timer);​
       batch.resolve({​
        votes: batch.collectedVotes,​
        timedOutShards: batch.expectedShards - batch.completedShards,​
       });​
     }​

   this.pending.clear();​
   await Promise.all(this.workers.map((w) => w.terminate()));​
  }​
}​
