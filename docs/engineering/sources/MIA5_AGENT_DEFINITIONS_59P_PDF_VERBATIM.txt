const ABSTAIN: RawAgentSignal = {
  action: 'ABSTAIN',
  rawConfidence: 0,
  expectedEdgeBps: 0,
  reasonCode: 'NO_TRIGGER',
};

function clamp(val: number, min: number, max: number): number {
  return Math.max(min, Math.min(max, val));
}

function f(ctx: AgentContextSnapshot, slot: FeatureSlot, fallback = 0): number {
  const v = ctx.features[slot];
  return Number.isFinite(v) ? v : fallback;
}
